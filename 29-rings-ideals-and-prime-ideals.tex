\section*{Rings, Ideals, and Prime Ideals}
% BLOG-CONTENT-BEGIN
Let's begin with two fundamental objects: rings and modules.
\subsection{Rings and ideals}
\setcounter{definition}{0}\begin{definition}
A ring $R$ satisfies the following conditions:
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $(R,+)$ is an abelian group.
\setcounter{enumi}{1}\item For all $x,y,z\in R$, $(xy)z=x(yz)$, $x(y+z)=xy+xz$, and $(y+z)x=yx+zx$ (associativity and distributivity).
\setcounter{enumi}{2}\item $xy=yx$ (commutativity).
\setcounter{enumi}{3}\item There exists $1\in R$ such that $1x=x1=x$ (identity).
\end{enumerate}
If $S\subseteq R$ satisfies (i)--(iv), then $S$ is a \emph{subring} of $R$.
\end{definition}
\setcounter{example}{0}\begin{example}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $(\Z,+,\times)$, the ring of integers.
\setcounter{enumi}{1}\item $(\kk,+,\times)$, a field.
\setcounter{enumi}{2}\item $(\kk[x_1,\ldots,x_n],+,\times)$, the ring of polynomials over $\kk$.
\setcounter{enumi}{3}\item $(\{0\},+,\times)$, the zero ring.
\end{enumerate}
\end{example}
% Source page 3
\setcounter{definition}{1}\begin{definition}
For rings $A$ and $B$, a \emph{ring homomorphism} $f:A\to B$ is a map satisfying
\[
 f\in\Hom_{\Ab}(A,B),\qquad f(xy)=f(x)f(y),\qquad f(1)=1.
\]
\end{definition}
\setcounter{example}{1}\begin{example}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $\Z\hookrightarrow\Q$, the field of rationals.
\setcounter{enumi}{1}\item For $P\in\kk^n$, the evaluation map $\kk[x_1,\ldots,x_n]\to\kk$, $f\mapsto f(P)$, is a ring homomorphism.
\setcounter{enumi}{2}\item If $f:A\to B$ and $g:B\to C$ are ring homomorphisms, then $g\circ f:A\to C$ is a ring homomorphism.
\end{enumerate}
\end{example}
\begin{remark}
Denote by $\CRing$ the category of commutative rings. Then:
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $\Z$ is an \emph{initial object} of $\CRing$: for $A\in\Ob(\CRing)$ there exists a unique $f:\Z\to A$.
\setcounter{enumi}{1}\item $\{0\}$ is the \emph{terminal object}: for a given $A$ there exists a unique $g:A\to\{0\}$.
\setcounter{enumi}{2}\item If $\{A_i\}_{i=1}^n\subseteq\Ob(\CRing)$, then there exists $A\in\Ob(\CRing)$, called the \emph{product},
\[
A=\prod_{i=1}^n A_i,
\]
with pointwise addition and multiplication, and $(1,\ldots,1)$ as identity. It comes equipped with projections $P_i:A\to A_i$. For any collection of morphisms $g_i:B\to A_i$,
% Source page 4
there exists a unique $f:B\to A$ such that the diagram below commutes.
\begin{figure}[H]\centering
\begin{tikzpicture}[>=Stealth, every node/.style={font=\large},blue/.style={draw=blue!65!black}]

\node (B) at (6.3,2) {$B$}; \node (A) at (6.3,-2) {$A$};

Of all rings $B$ mapping to the $A_i$, $A$ is ``universal''.
\setcounter{enumi}{3}\item Direct sums need not be coproducts in $\CRing$. Coproducts nevertheless exist: for two commutative rings, the coproduct of $A_1$ and $A_2$ is $A_1\otimes_{\Z}A_2$.
Note that if $A_i\in\Ob(\mathsf{AbGps})$, then $\bigoplus_{i=1}^n A_i$ exists; it is universal among maps $(A_i\to B)\in\Hom_{\Ab}(A_i,B)$. If the $A_i$ are decorated with ring structure $(A_i,\times,1)$, then
\[
(1\in A_i)\longmapsto\underbrace{(0,0,\ldots,1,\ldots,0)}_{a_i}\in\bigoplus A_i.
\]
So $a_i=1$ for all $i$, which is absurd.
\end{enumerate}
\end{remark}
\setcounter{definition}{2}\begin{definition}
For an ideal $\aa\subseteq A$, the quotient $A/\aa$ is an abelian group, and its multiplication is well defined because $A\aa\subseteq\aa$.
% The handwritten definition at the foot of p.4 is overwritten; the legible quotient and absorption statements are retained.
% Source page 5
Indeed,
\[
(x+\aa)(y+\aa)=xy+x\aa+y\aa+\aa^2
=xy+(x+y+\aa)\aa=xy+\aa.
\]
\end{definition}
\begin{remark}
One can view ideals $\aa$ as subgroups of $A$ with an action of $A$, captured by
\[
\psi:A\longrightarrow\End_{\Ab}(\aa)\text{ is a ring homomorphism},\qquad
\psi(x)(a):=xa\quad(x\in A,\ a\in\aa).
\]

\end{remark}
\setcounter{example}{2}\begin{example}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item If $f\in\Hom_{\CRing}(A,B)$, then $\ker f$ is an ideal of $A$.
\setcounter{enumi}{1}\item For $a\in A$, $(a):=\{xa\mid x\in A\}$ is an ideal of $A$ generated by $a$. Similarly, for a subset $S\subseteq A$,
\[
(S):=\left\{\sum_{i\in F}x_i y_i\ \middle|\ x_i\in A,\ y_i\in S,\ F\text{ finite}\right\}
\]
is the ideal generated by $S$, that is, the smallest ideal in $A$ containing $S$.
\setcounter{enumi}{2}\item $(0)$ and $(1)$ are the only ideals of $\kk$. Hence, by (i), every morphism $f:\kk\to B$ with $B\ne0$ is injective.
\setcounter{enumi}{3}\item If $\{\aa_i\}_{i\in I}$ are ideals of $A$, then so are
\[
\sum_{i\in I}\aa_i:=\left\{\sum_{i\in F}a_i\ \middle|\ a_i\in\aa_i,\ F\subseteq I\text{ finite}\right\},
\qquad \bigcap_{i\in I}\aa_i,
\]
and $\prod_{i\in F}\aa_i$, the ideal generated by $\{\prod_{i\in F}a_i\mid a_i\in\aa_i\}$,
% Source page 6
where $F\subseteq I$ is finite.
\setcounter{enumi}{4}\item In $\Z$, all ideals are of the form $(n)$.
\setcounter{enumi}{5}\item If $\aa,\bb$ are ideals of $A$, then so is the \emph{ideal quotient}
\[
(\aa:\bb):=\{x\in A\mid x\bb\subseteq\aa\}.
\]
If $\aa=(0)$, then $(0:\bb)=:\Ann(\bb)$ is called the \emph{annihilator} of $\bb$.
\setcounter{enumi}{6}\item For an ideal $\aa$ of $A$, $\rad(\aa):=\{x\in A\mid x^n\in\aa\text{ for some }n\geq1\}$ is also an ideal of $A$, called the \emph{radical} of $\aa$, since nilpotent elements of $A/\aa$ form an ideal by the binomial theorem.
\setcounter{enumi}{7}\item Let $\aa$ be an ideal of $A$ and $\bb$ an ideal of $B$. For $f\in\Hom_{\CRing}(A,B)$, the ideal $(f(\aa))$ generated by $f(\aa)$ in $B$ is called the \emph{extension} of $\aa$ to $B$, denoted $\aa^e$. The ideal $f^{-1}(\bb)$ of $A$ is the \emph{contraction} of $\bb$ in $A$, denoted $\bb^c$.
\end{enumerate}
\end{example}
\subsection{Distribution of ideals of \texorpdfstring{$A$}{A}}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item Let $\Sigma$ be the set of all ideals in $A$, excluding $A$. This is a poset under inclusion. Since $0\in\Sigma$, $\Sigma\ne\varnothing$. If $\{\aa_i\}_{i\in I}$ is a chain of ideals, then $\bigcup_{i\in I}\aa_i\in\Sigma$ (as $1\notin\bigcup_i\aa_i$); furthermore, this union is an upper bound
% Source page 7
for the chain. By Zorn's lemma, $\Sigma$ has a maximal element.
Therefore every proper ideal $\aa$ is contained in a maximal ideal, as illustrated below: the same argument can be made in $A/\aa$.
\begin{figure}[H]\centering
\includegraphics[width=.84\textwidth]{ca1-fig-02.pdf}
\end{figure}
\setcounter{enumi}{1}\item If $\mm$ is a maximal ideal of $A$, then
\[
A\setminus\mm=\{x\in A\mid (x)+\mm=A\}.
\]
If $A\setminus\mm=\{x\in A\mid x^{-1}\in A\}$, then $\mm$ is the unique maximal ideal. Such a ring is called \emph{local}. Since $\mm$ is maximal, $A/\mm$ has no proper ideals except $(0)$; hence $A/\mm$ is a field.
% Source page 8
\setcounter{enumi}{2}\item If $A\ne0$, let
\[
\begin{aligned}
0\in D_0&:=\{x\in A\mid xy=0\text{ for some }0\ne y\in A\}
&&\text{(zero-divisors)},\\
1\in D_1&:=\{x\in A\mid xy=1,\ y\in A\}
&&\text{(one-divisors, or units)}.
\end{aligned}
\]
Clearly, $D_0\cap D_1=\varnothing$. An element belongs either to $D_0$, to $D_1$, or to $D_0^c\cap D_1^c$.
\begin{figure}[H]\centering
\includegraphics[width=.42\textwidth]{ca1-fig-03.pdf}
\end{figure}
If $A\setminus\{0\}=D_1$, then $A$ is a field. If $A\setminus\{0\}=D_0^c$, then $A$ is an \emph{integral domain}.
\setcounter{enumi}{3}\item If, for an ideal $\pp$ of $A$, $A/\pp$ is an integral domain, then $\pp$ is a \emph{prime ideal}; equivalently, $xy\in\pp\Rightarrow x\in\pp$ or $y\in\pp$.
\end{enumerate}
\setcounter{theorem}{0}\begin{lemma}[Prime avoidance]
If $\{\pp_i\}_{i=1}^n$ are prime ideals and $\aa$ is an ideal with $\aa\subseteq\bigcup_{i=1}^n\pp_i$, then $\aa\subseteq\pp_i$ for some $i$.
\end{lemma}
\begin{proof}
The case $n=1$ is clear. Assume the result for $n-1$. For $1\le i\le n$, if $x\in\aa\Rightarrow x\in\pp_j$ for $j\ne i$, then $\aa\subseteq\bigcup_{j\ne i}\pp_j$. By the induction hypothesis, $\aa\subseteq\pp_j$ for some $j$. Therefore, for every $1\le i\le n$, there exists $x_i\in\aa$ such that $x_i\notin\pp_j$ for $j\ne i$, so $x_i\in\pp_i$. Now let
\[
y=\sum_{i=1}^n x_1\cdots x_{i-1}x_{i+1}\cdots x_n.
\]
Then $y\in\aa$, but $y\notin\bigcup_i\pp_i$, a contradiction.
\end{proof}
% Source page 9
Consequently, an ideal cannot be covered by finitely many prime ideals unless it is contained in one of them, as depicted here.

\setcounter{theorem}{1}\begin{lemma}[Factoring]
If $\{\aa_i\}_{i=1}^n$ are ideals and $\pp$ is a prime ideal such that $\pp\supseteq\bigcap_{i=1}^n\aa_i$, then $\pp\supseteq\aa_i$ for some $i$.
\end{lemma}
\begin{proof}
If $\aa_i\nsubseteq\pp$ for every $i$, choose $x_i\in\aa_i$ with $x_i\notin\pp$. But $\prod_{i=1}^n x_i\in\bigcap_i\aa_i\subseteq\pp$, a contradiction.
\end{proof}
\begin{remark}
In particular, if $\pp=\bigcap_{i=1}^n\aa_i$, then $\pp=\aa_i$ for some $i$. Thus a prime ideal cannot be a nontrivial finite intersection of strictly larger ideals.
\end{remark}
\setcounter{theorem}{2}\begin{proposition}
\[\rad((0))=\bigcap_{\pp\text{ prime}}\pp.\]
\end{proposition}
\begin{proof}
If $f\in\rad((0))$, then $f^n=0\in\pp$, so $f\in\pp$ and hence $f\in\bigcap_{\pp\text{ prime}}\pp$.
If $f\notin\rad((0))$, let $\Sigma$ be the poset of all ideals $\aa$ such that $f^n\notin\aa$ for every $n$. As $0\in\Sigma$, this poset is nonempty. If $\{\aa_i\}_{i\in I}$ is a chain, then $\bigcup_i\aa_i\in\Sigma$ and bounds the chain. By Zorn's lemma, $\Sigma$ has a maximal element $\mm$.
\emph{Claim:} $\mm$ is prime.
% Source page 10
Let $x,y\notin\mm$. Then $\mm+(x),\mm+(y)\notin\Sigma$ by maximality. Therefore, $f^a\in\mm+(x)$ and $f^b\in\mm+(y)$, so $f^{a+b}\in\mm+(xy)$ and thus $xy\notin\mm$. Therefore $f\notin\mm$, implying $f\notin\bigcap_{\pp\text{ prime}}\pp$.
\end{proof}
