\section*{Problems for Anant Mudgal}
% BLOG-CONTENT-BEGIN
\begin{enumerate}[label=P\arabic*.]
\item[$\mathrm{P1}^{***}$.] Let $a,b,c\in\mathbb R$ satisfy
$\min(a,b,c)=1/2$ and $a+b+c=3$, and let $v^2-3v+1=0$.
Prove $\sum_{\mathrm{cyc}}(a^2-bc+1)/(bc+va-1)\geq0$. \emph{Inequality.}
\item[$\mathrm{P2}^{*}$.] Prove the continued-fraction identity
\[
\left(\vcenter{\hbox{$\displaystyle 1+\dfrac{1+\dfrac{1+\dfrac{1+\dfrac{1+\cdots}{4}}{3}}{2}}{1}$}}\right)^2
=\vcenter{\hbox{$\displaystyle 2+\dfrac{4+\dfrac{8+\dfrac{16+\dfrac{32+\cdots}{4}}{3}}{2}}{1}$}}.
\]
\emph{Algebra.}
\setcounter{enumi}{2}\item Find all continuous $f:(1,\infty)\to\mathbb R$ such that
$f(x+y)f(x)f(y)=f(x+y)(f(x)+f(y))-f(x)f(y)$.
\emph{Functional equation.}
\setcounter{enumi}{3}\item Prove that
$\{\pi\}=\sqrt{32+\sqrt{512+\sqrt{8192+\cdots}}}
(1/15+1/63+1/143)-1$. The continuation of the nested radical is not
specified, so the expression needs a general term before it defines a problem.
\emph{Computations.}
\item Among the numbers in $[100000,200000]$, there exists a prime gap $\ll12$.
\emph{Number theory.}

The comparison sign $\ll$ needs a specified meaning here before the claim has a
definite numerical bound.
\item Given any set of $n$ points, $n\geq3$, define a $k$-rich line to be one containing exactly $k$ points.
Prove that the number of $2$-rich and $3$-rich lines counted together is more than ``$2$ or $3$''.
\emph{Geometry.}

The intended threshold and the needed non-collinearity hypothesis have not been
specified in the statement.
% Source page 20 - right
\setcounter{enumi}{6}\item Given $\sum_{n=1}^\infty1/n^2=\pi^2/6$, compute the probability that a number $n$ is square-free,
meaning that it has no square divisor other than $1$. \emph{Probability theory.}
\item For positive $a,b,c$ with $\sum_{\mathrm{cyc}}1/(a+b+1)=1$, prove
$\sqrt[4]{\prod_{\mathrm{cyc}}(a^2-ab+b^2)}+abc\geq2$. \emph{Open!}
\item Under the same constraint, prove
\[
\sqrt[4]{\prod_{\mathrm{cyc}}(a^2-ab+b^2)}
+\frac{2abc}{\sqrt{a+b+c+1}}\geq\sqrt{a+b+c+1}.
\]
I forgot the solution!
\end{enumerate}
