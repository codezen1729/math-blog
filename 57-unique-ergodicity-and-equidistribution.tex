\section*{Unique Ergodicity and Equidistribution}
% BLOG-CONTENT-BEGIN
Birkhoff's theorem connects ergodicity with the distribution of orbits. The
converse question leads to a surprisingly rigid condition: unique ergodicity.

\setcounter{theorem}{22}\begin{theorem}[Unique ergodicity]\label{thm:unique-ergodicity}
The convergence
\[
  A_N^f(x):=\frac1N\sum_{n=0}^{N-1}f(T^nx)\longrightarrow C_f,
  \qquad f\in C(X),
\]
where $C_f$ is independent of $x$, is equivalent to
$\lvert\mathcal M^T(X)\rvert=1$.
\end{theorem}

\begin{proof}
$(\Downarrow)$ Choose some $\delta_x\in\mathcal M(X)$. By the existence theorem
for invariant measures,
\[
  \frac1N\sum_{n=0}^{N-1}\delta_{T^nx}\overset{*}{\longrightarrow}\mu,
\]
for every accumulation point is $T$-invariant and there cannot be more than
one, as $\lvert\mathcal M^T(X)\rvert=1$. Therefore
\[
  \frac1N\sum_{n=0}^{N-1}f(T^nx)\longrightarrow\int_X f\,d\mu,
\]
independently of $x$.

% Source page 2
$(\Uparrow)$ Let $\mu\in\mathcal M^T(X)$. As $\mu$ is $T$-invariant, we have
$\int_X f(T^nx)\,d\mu=\int_X f\,d\mu$. By the dominated convergence theorem,
\[
  \int_X f\,d\mu
  =\int_X\lim_{N\to\infty}\frac1N\sum_{n=0}^{N-1}f(T^nx)\,d\mu
  =\int_X C_f\,d\mu=C_f.
\]
Thus $\int_X f\,d\mu=C_f$ for all $\mu\in\mathcal M^T(X)$, and hence
$\lvert\mathcal M^T(X)\rvert=1$.
\end{proof}

\begin{remark}[1]
Notice that Choquet's theorem implies that $\lvert\mathcal M^T(X)\rvert=1$
if $\lvert\mathcal E^T(X)\rvert=1$, for $\mathcal M^T(X)$ consists of
``convex linear combinations'' of elements of $\mathcal E^T(X)$.
\end{remark}

\begin{remark}[2]
When $(X,T)$ is uniquely ergodic, the convergence in the theorem above is
uniform. Otherwise, for some $\varepsilon>0$ there exist points $x_k$ and
integers $N_k\to\infty$ such that
\[
  \left|\frac1{N_k}\sum_{n=0}^{N_k-1}g(T^nx_k)-C_g\right|\geq\varepsilon.
\]
Set $\mu_k=\frac1{N_k}\sum_{n=0}^{N_k-1}\delta_{T^nx_k}$. By weak-$*$ compactness,
some subsequence converges to $\nu$, and therefore
$\left|\int_X g\,d\nu-C_g\right|\geq\varepsilon$. Now
$C_g=\int_X g\,d\mu$, where $\mu\in\mathcal M^T(X)$, so $\nu\ne\mu$.
This is a contradiction.

Thus the convergence is uniform in $x$; the orbit averages are
well-distributed.
\end{remark}

% Source page 3
This motivates the following definition of equidistribution for a sequence $(x_j)_j$.

\setcounter{definition}{9}\begin{definition}[Equidistribution]
We say that $(x_n)_n$, with $x_n\in X$, is \emph{equidistributed} in $(X,\mu)$
if, for every $f\in C(X)$, we have
\[
  \lim_{n\to\infty}\frac1n\sum_{j=1}^n f(x_j)
  =\int_X f(x)\,d\mu(x).
\]
\end{definition}

\begin{remark}
Unique ergodicity links equidistribution in $X$ with invariant measures in $\mathcal M(X)$.
\end{remark}

\setcounter{definition}{10}\begin{definition}[Generic point]
We say that $x\in X$ is \emph{generic for $\mu$} if either of the following
equivalent conditions holds:
\begin{enumerate}[label=(\roman*)]
  \item $\displaystyle\frac1n\sum_{j=1}^n\delta_{T^jx}
  \overset{*}{\longrightarrow}\mu$ in $\mathcal M(X)$;
  \item $\displaystyle\lim_{n\to\infty}\frac1n\sum_{j=1}^n f(T^jx)
  =\int_Xf\,d\mu$ for every $f\in C(X)$.
\end{enumerate}
\end{definition}

The following proposition is an immediate consequence of the ergodic theorem.

\setcounter{theorem}{23}\begin{proposition}
If $\mu\in\mathcal M^T(X)$ is ergodic, then almost every $x\in X$ is generic.
\end{proposition}

\begin{proof}
By the Stone--Weierstrass theorem, $(C(X),\lVert\cdot\rVert_\infty)$ is
separable. Let
% Source page 4
$(f_n)_n$ be dense in $C(X)$. We use a dense subset because
\[
  \lim_{n\to\infty}\frac1n\sum_{j=1}^n f(T^jx)
  =\int_X f\,d\mu\qquad(\forall x\in X_f),\qquad\mu(X_f)=1.
\]
But $X_f$ depends on $f$; we now need to drop that dependence.

For each $f_n$, one obtains a set $X_{f_n}$, and
$\mu\bigl(\bigcap_nX_{f_n}\bigr)=1$. Set $X'=\bigcap_nX_{f_n}$. By the
ergodic theorem, for all $x\in X'$ and $i\geq1$,
\[
  \frac1N\sum_{n=0}^{N-1}f_i(T^nx)\longrightarrow\int_X f_i\,d\mu.
\]
As $(f_i)_i$ is dense in $(C(X),\lVert\cdot\rVert_\infty)$, it is dense
uniformly, and for each $f\in C(X)$ and $\varepsilon>0$ we can choose $i$ so that
\[
  \|f-f_i\|_\infty<\varepsilon.
\]
For this $i$ we have
(with $A_N^f$ denoting the average)
\begin{align*}
  &\left|\int f\,d\mu-\limsup_{N\to\infty}
    \frac1N\sum_{n=0}^{N-1}f(T^nx)\right|
  =\left|\int f\,d\mu-\limsup_{N\to\infty}A_N^f\right|\\
  &\quad=\left|\int f\,d\mu-\limsup_{N\to\infty}A_N^f
    -\limsup_{N\to\infty}A_N^{f_i}
    +\limsup_{N\to\infty}A_N^{f_i}
    -\int f_i\,d\mu+\int f_i\,d\mu\right|\\
  &\quad\leq\left|\int f\,d\mu-\int f_i\,d\mu\right|
    +\left|\limsup_{N\to\infty}(A_N^f-A_N^{f_i})\right|
    +\left|\int f_i\,d\mu-\limsup_{N\to\infty}A_N^{f_i}\right|\\
  &\quad<2\varepsilon.
\end{align*}
One can similarly estimate $\liminf_{N\to\infty}A_N^f$ to obtain that,
for all $x\in X'$, $\lim_{N\to\infty}A_N^f(x)=\int_X f\,d\mu$.
\end{proof}

% Source page 5
For a probabilistic interpretation, consider $[0,1]$ with Lebesgue measure.

\setcounter{theorem}{24}\begin{lemma}
Let $[0,1]$, equipped with Lebesgue measure, be our setting. If $(x_n)_{n\geq1}$
is equidistributed, then
\[
  \lim_{n\to\infty}\frac1n
  \bigl|\{j:1\leq j\leq n,\ x_j\in[a,b]\}\bigr|=b-a.
\]
\end{lemma}

In other words, equidistribution means being uniformly distributed over
$[0,1]$. The same interpretation applies to ergodic orbits in a probability
space. These sequences correspond to ergodic measures, the extreme points of
$\mathcal M^T(X)$.

\Needspace{7cm}
\begin{proof}

Let $f^-$ and $f^+$ be as shown in the figure. Note that
$f^-\leq\chi_{[a,b]}\leq f^+$, and one can arrange $f^\pm$ so that
\[
  \int_0^1(f^+-f^-)(x)\,dx\leq2\varepsilon.
\]

% Source page 6
Using the same approximation argument,
\begin{align*}
  b-a-2\varepsilon
  &\leq\int_0^1f^-\,dx
  \leq\liminf_{N\to\infty}A_N^{\chi_{[a,b]}}
  \leq\limsup_{N\to\infty}A_N^{\chi_{[a,b]}}\\
  &\leq\int_0^1f^+\,dx\leq b-a+2\varepsilon.
\end{align*}
Thus $\liminf_{N\to\infty}A_N^{\chi_{[a,b]}}
=\limsup_{N\to\infty}A_N^{\chi_{[a,b]}}=b-a$.
\end{proof}

\begin{remark}
Therefore, equidistribution theorems can be derived from ergodic
theory, provided one finds the right transformation that is uniquely
ergodic. For instance, the orbit sequence $\{n\alpha\}$ modulo $1$, with
$\alpha\notin\mathbb Q$, is equidistributed in $[0,1]$, for $R_\alpha$ is
uniquely ergodic. Weyl's theorem describes equidistribution modulo $1$ of a
polynomial sequence $P(n)$ when at least one nonconstant coefficient is
irrational.
\end{remark}

\setcounter{theorem}{25}\begin{theorem}[Jewett--Krieger]
\[
  \begin{gathered}
    \text{Ergodic system on a}\\
    \text{Lebesgue space}\\
    (X,\mathcal B,\mu,T)
  \end{gathered}
  \quad\cong\quad
  \begin{gathered}
    \text{Minimal uniquely ergodic}\\
    \text{topological dynamical system}
  \end{gathered}
\]
Here \emph{minimal} means that, for every $x$, the orbit of $x$ is dense in $X$.
\end{theorem}

Note that measurable isomorphisms only preserve sizes of sets. If
% Source page 7
our $(X,\mathcal B,\mu,T)$ were an ergodic topological dynamical system,
then the isomorphism does not preserve any structure on $\mathcal M^T(X)$,
because, in the image, $\lvert\mathcal M^T(X)\rvert=1$ regardless of what
$\mathcal M^T(X)$ is.

Thus this invariant-measure structure is topological rather than purely measurable.

Ergodic decomposition addresses one side of the picture. The other is the
classification of systems up to measurable isomorphism.
