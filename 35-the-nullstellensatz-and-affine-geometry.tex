\subsection{Hilbert's Nullstellensatz}
% Source page 5
\setcounter{definition}{14}\begin{definition}
A ring $R$ is \emph{Jacobson} if, for every prime $\pp$,
\[\pp=\bigcap_{\substack{\mm\supseteq\pp\\\mm\text{ maximal}}}\mm.\]
\end{definition}
\setcounter{example}{8}\begin{example}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item A field $\kk$ is a Jacobson ring.
\setcounter{enumi}{1}\item $\Z$ is Jacobson: every nonzero prime ideal is maximal, and $(0)=\operatorname{nilradical}(\Z)=\bigcap_{\mm\text{ maximal}}\mm=\bigcap_{\pp\text{ prime}}\pp$.
\setcounter{enumi}{2}\item $\kk[x]$ is Jacobson. It is a PID by the division algorithm. Every nonzero prime is generated by an irreducible polynomial and is therefore maximal. Since $\kk[x]$ is a domain, as in (ii), $(0)=\bigcap_{\mm\text{ maximal}}\mm=\bigcap_{\pp\text{ prime}}\pp$.
\end{enumerate}
\end{example}
The following lemma characterizes Jacobson rings.
\setcounter{theorem}{23}\begin{lemma}
$R$ is Jacobson if and only if, for prime $\pp$ and $0\ne b\in R/\pp$, whenever $(R/\pp)_{(b)}$ is a field, $R/\pp$ is a field.
\end{lemma}
\begin{proof}
Suppose $R$ is Jacobson. Then $S:=R/\widetilde\pp$ is Jacobson for prime $\widetilde\pp$. In $S$,
\[(0)=\bigcap_{\pp\text{ prime}}\pp=\bigcap_{\mm\text{ maximal}}\mm.\tag{*}\]
Suppose $S_{(b)}$ is a field. If $\pp$ is a nonzero prime ideal and $\pp\cap\{1,b,b^2,\ldots\}=\varnothing$, then $(b)^{-1}\pp$ is a nonzero proper ideal of $S_{(b)}$, absurd. Therefore every nonzero prime $\pp$ meets $\{1,b,b^2,\ldots\}$; but (*) gives $b=0$, again absurd. Thus $(0)$ is maximal and $S$ is a field.
% Source page 6
Conversely, let $\qq<R$ be prime and set $I=\bigcap_{\mm\supseteq\qq,\ \mm\text{ maximal}}\mm$. We must show $I=\qq$. Suppose not, and choose $f\in I\setminus\qq$. Set
\[\Sigma=\{\pp\text{ prime in }R\mid\pp\supseteq\qq,\ f\notin\pp\},\]
ordered by inclusion. By Zorn's lemma there is a maximal $\widetilde\pp\in\Sigma$. Clearly $\widetilde\pp$ is not a maximal ideal, as $f$ is in every maximal ideal containing $\qq$. Hence $S:=R/\widetilde\pp$ is not a field, but $S_{(f)}$ is a field, since $\widetilde\pp$ is maximal among the prime ideals disjoint from the powers of $f$. This is a contradiction.
\end{proof}
\setcounter{theorem}{24}\begin{theorem}[Hilbert's Nullstellensatz, Jacobson form]
Let $R$ be a Jacobson ring and $S$ a finitely generated $R$-algebra. Then:
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $S$ is a Jacobson ring.
\setcounter{enumi}{1}\item If $\nn<S$ is maximal, then $\mm:=\nn\cap R$ is maximal in $R$, and $S/\nn$ is a finite field extension of $R/\mm$.
\end{enumerate}
\end{theorem}
\setcounter{theorem}{25}\begin{lemma}
Let $A\subseteq B$ be domains, with $B$ integral over $A$. Then $B$ is a field if and only if $A$ is a field.
\end{lemma}
\begin{proof}
Let $A$ be a field and $0\ne y\in B$. Choose a monic relation $y^n+a_1y^{n-1}+\cdots+a_n=0$, $a_i\in A$, of least degree. As $B$ is an integral domain, $a_n\ne0$, so $a_n$ is a unit. Hence
\[
y\bigl(-a_n^{-1}(y^{n-1}+a_1y^{n-2}+\cdots+a_{n-1})\bigr)=1.
\]
Thus $y^{-1}=-a_n^{-1}(y^{n-1}+a_1y^{n-2}+\cdots+a_{n-1})$,
% Source page 7
and $B$ is a field. Conversely, if $B$ is a field and $0\ne x\in A$, then $x^{-1}\in B$ is integral over $A$. Thus $x^{-m}+a_1x^{-m+1}+\cdots+a_m=0$, with $a_i\in A$. Therefore
\[
x^{-1}=-(a_1x^{-m+1}+\cdots+a_m)x^{m-1}
=-(a_1+a_2x+\cdots+a_mx^{m-1})\in A.
\]
Hence $A$ is a field.
\end{proof}
\setcounter{theorem}{26}\begin{corollary}
If $\qq<B$ is prime, then $\qq$ is maximal if and only if $\qq^c=\qq\cap A<A$ is maximal.
\end{corollary}
\begin{proof}
$B/\qq$ is integral over $A/\qq^c$. Since $\qq,\qq^c$ are prime, the two quotients are domains.
The preceding lemma now gives the equivalence.
\end{proof}
\begin{proof}[Proof of the Jacobson-form Nullstellensatz]
\emph{Strategy.} To show $S$ is Jacobson, we must show that, for prime $\pp<S$ and $0\ne b\in S/\pp$, if $(S/\pp)_{(b)}$ is a field, then $S/\pp$ is a field. The ring map $R\to S$ induces $R/\pp^c\hookrightarrow S/\pp$. Both quotients are domains. It therefore suffices to show that, for domains $R\hookrightarrow S$, if $b\in S$ and $S_{(b)}$ is a field, then $S$ is a field. We will also show that $R$ is a field and $S$ is a finite extension of $R$, thereby proving both assertions at once. Proceed by induction on the number of generators of the algebra $S$.
% Source page 8
\emph{Base case.} Suppose $S$ is generated by one element $t$. There is a map $R[x]\to S$, sending $x$ to $t$, so $S=R[x]/\mathfrak g$. If $\mathfrak g=0$, then some $b\in R[x]$ makes $R[x]_{(b)}$ a field. Hence $K[x]_{(b)}$ is a field, where $K$ is the field of fractions of $R$. But $K[x]$ is a Jacobson domain, so the lemma, applied to $(0)$, would make $K[x]$ a field, absurd.

Thus $\mathfrak g\ne0$. Choose $p(x)\in\mathfrak g$, $p(x)=p_nx^n+\cdots+p_0$. In $S$ one has $p_nt^n+\cdots+p_0=0$. Consequently, $S_{(p_n)}$ is integral over $R_{(p_n)}$, so $b/p_n$ is integral over $R_{(p_n)}$. Clearing denominators gives $q_mb^m+\cdots+q_0=0$. Since $S$ is a domain, divide by enough powers of $b$ to assume $q_0\ne0$. Then $S_{(b)}$ is integral over $R_{(p_nq_0)}$, since
\[
0=\frac{q_mb^m+\cdots+q_0}{q_0b^m}
=\left(\frac1b\right)^m+\frac{q_1}{q_0}\left(\frac1b\right)^{m-1}+\cdots+\frac{q_m}{q_0}.
\]
Because $S_{(b)}$ is a field, $R_{(p_nq_0)}$ is a field. Since $R$ is a Jacobson domain, $R$ is a field. As $S=R[x]/\mathfrak g$, $S$ is a finite extension of $R$.
% Source page 9
\emph{Inductive step.} If $S$ is an $R$-algebra generated by $r$ elements, let $S'$ be the $R$-subalgebra generated by $r-1$ of them. By induction $S'$ is Jacobson, and so is $S$ by the base case. If $\nn$ is maximal in $S$, then
\[
R/(R\cap\nn)\xhookrightarrow{\ 1\ }S'/(S'\cap\nn)
\xhookrightarrow{\ 2\ }S/\nn,
\]
where the arrows labelled $1$ and $2$ are finite field extensions.
\end{proof}
\setcounter{theorem}{27}\begin{corollary}[Weak Nullstellensatz]
Let $\kk$ be algebraically closed. Every maximal ideal of $\kk[x_1,\ldots,x_n]$ has the form $\mm_P=(x_1-a_1,\ldots,x_n-a_n)$ for $P=(a_1,\ldots,a_n)\in\kk^n$.
\end{corollary}
\begin{proof}
Since $\kk[x_1,\ldots,x_n]/\mm_P\cong\kk$, $\mm_P$ is maximal. If $\nn$ is maximal in $\kk[x_1,\ldots,x_n]$, then $\kk[x_1,\ldots,x_n]/\nn$ is algebraic over $\kk/(\nn\cap\kk)=\kk$. Since $\kk$ is algebraically closed, $\kk[x_1,\ldots,x_n]/\nn=\kk$. Let $a_i=[x_i]$ in this quotient and $P=(a_1,\ldots,a_n)$. Then $\mm_P\subseteq\nn$. Since $\mm_P$ is maximal, $\nn=\mm_P$.
\end{proof}
\subsection{A little geometry}
Consider the two objects $\kk^n$ and $\kk[x_1,\ldots,x_n]$. To $I\subseteq\kk[x_1,\ldots,x_n]$ associate the set
\[Z(I):=\{P\in\kk^n\mid f(P)=0\text{ for every }f\in I\}.\]
% Source page 10
Clearly $Z(I)=Z((I))$. Sets of this form are \emph{algebraic subsets} of $\kk^n$. For ideals $\{J_i\}_i$ of $\kk[x_1,\ldots,x_n]$,
\[
\bigcap_i Z(J_i)=Z\left(\bigcup_i J_i\right),\qquad
\bigcup_{i=1}^n Z(J_i)=Z\left(\prod_{i=1}^n J_i\right).
\]
On an algebraic set $X$, declaring algebraic subsets of $X$ to be closed defines the \emph{Zariski topology}.

For a subset $X\subseteq\kk^n$, set $I(X):=\{f\in\kk[x_1,\ldots,x_n]\mid f(P)=0\text{ for every }P\in X\}$. Clearly $I(X)$ is an ideal. For algebraic $X$, define its \emph{coordinate ring}, also called an \emph{affine ring}, by $A(X):=\kk[x_1,\ldots,x_n]/I(X)$.

If $f\in A(X)$ and $f^n\equiv0$, then $f^n(P)=0$ for all $P\in X$. Since $\kk$ is a domain, $f\equiv0$. Thus $\{0\}$ is the only nilpotent element in $A(X)$, and $I(X)=\rad(I(X))$. Such ideals are called \emph{radical ideals}.
\setcounter{theorem}{28}\begin{theorem}[Strong Nullstellensatz]
Let $\kk$ be algebraically closed and $I$ an ideal in $\kk[x_1,\ldots,x_n]$. Then
\[I(Z(I))=\rad(I).\]
\end{theorem}
\begin{proof}
Since all maximal ideals of $\kk[x_1,\ldots,x_n]$ are of the form $\mm_P$, and these are maximal, points of $\kk^n$ correspond to maximal ideals of $\kk[x_1,\ldots,x_n]$. Hence
\[
I(Z(I))=\bigcap_{P\in Z(I)}\mm_P
=\bigcap_{\substack{\mm\supseteq I\\\mm\text{ maximal}}}\mm
% Source page 11
=\bigcap_{\substack{\pp\supseteq I\\\pp\text{ prime}}}\pp
=\rad(I),
\]
since $\kk[x_1,\ldots,x_n]$ is Jacobson.
\end{proof}
\begin{remark}
Clearly $Z(I(X))=X$. Thus we have the correspondences
\[
\begin{aligned}
\text{algebraic sets}&\longleftrightarrow\text{radical ideals}\\
&\longleftrightarrow\text{quotients of }\kk[x_1,\ldots,x_n]\text{ by radical ideals}\\
&\longleftrightarrow\text{coordinate rings}.
\end{aligned}
\]
\end{remark}
\setcounter{definition}{15}\begin{definition}
Let $X\subseteq\kk^m$ and $Y\subseteq\kk^n$ be algebraic sets. A map
\[
F:\kk^m\to\kk^n,\qquad
(a_1,\ldots,a_m)\mapsto(f_1(a_1,\ldots,a_m),\ldots,f_n(a_1,\ldots,a_m)),
\]
where $f_i\in\kk[x_1,\ldots,x_m]$, is a \emph{morphism of $X$ to $Y$} if $F|_X:X\to Y$.
\end{definition}
Given $F:X\to Y$, we have
\[
F^{\#}:\kk[y_1,\ldots,y_n]\longrightarrow\kk[x_1,\ldots,x_m],
\qquad y_i\longmapsto f_i(x_1,\ldots,x_m).
\]
This induces $F^{\#}:A(Y)\to A(X)$. With $X\mapsto A(X)$ and $F\mapsto F^{\#}$, we obtain an equivalence of categories:
% Source page 12
\[
\boxed{\text{algebraic sets}\ \longleftrightarrow\ \bigl(\text{finitely generated reduced }\kk\text{-algebras}\bigr)^{\mathrm{op}}}
\]
This gives a bridge between algebra and geometry.
