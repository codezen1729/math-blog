\section*{Attracting and Repelling Fixed Points}
% BLOG-CONTENT-BEGIN
Let $f\colon S\to S$ be a holomorphic self-map of a Riemann surface. Fixed points provide local models for its dynamics, while the distribution of critical points governs how far those models extend, especially for rational maps on $\widehat{\mathbb C}$.

Understanding the dynamics near fixed points amounts to finding a change of coordinates in which the orbits of $f$ are easy to describe. A general technique is to construct a local biholomorphism $\phi$ such that the following diagram commutes.
\begin{figure}[H]\centering
\includegraphics[width=.32\textwidth]{cd2-fig-01.pdf}
\FigureTag{CD2.1}{fig:cd2-01}
\end{figure}
% Source page 2
That is, $\phi\circ f\circ\phi^{-1}=g$ locally around $0$, where $g$ is preferably a polynomial. One begins by finding maps $\{\phi_k\}_k$ such that $\phi_k\circ f=g\circ\phi_{k+1}$, then proves that $\phi_k\to\phi$ uniformly. Consequently,
\[
\lim_{k\to\infty}(\phi_k\circ f)=\lim_{k\to\infty}(g\circ\phi_{k+1}),
\quad\phi\circ f=g\circ\phi,
\quad\phi\circ f\circ\phi^{-1}=g.
\]
\setcounter{definition}{6}\begin{definition}
A fixed point is \emph{superattracting} if its multiplier $\lambda$ is $0$, geometrically attracting if $0<|\lambda|<1$, and repelling if $1<|\lambda|$.
\end{definition}
We deal with all fixed points except those for which $\lambda\in S^1$. First let us understand the dynamics locally, and then describe the phenomenon governing it.
\subsection{Geometrically attracting and repelling points}
For $f\in\Hol(S,S)$, let $\varphi\circ f\circ\varphi^{-1}=\widehat f$ be its coordinate representation in a chart centred at a fixed point. Here $|\widehat f'(0)|\notin\{0,1\}$. On sufficiently small domains, $\widehat f$ behaves like $\widehat f'(0)z=\lambda z$. If $|\lambda|<1$, $f$ pulls; if $|\lambda|>1$, $f$ pushes. More precisely:
% Source page 3
\Needspace{18\baselineskip}
\setcounter{theorem}{10}\begin{theorem}[Koenigs linearisation]\label{cd2:koenigs}
If the multiplier satisfies $|\lambda|\notin\{0,1\}$, there is a map $\phi$, unique up to multiplication by a nonzero constant, such that the following diagram commutes.
\begin{figure}[H]\centering
\includegraphics[width=.32\textwidth]{cd2-fig-02.pdf}
\FigureTag{CD2.2}{fig:cd2-02}
\end{figure}
\end{theorem}
\begin{proof}
\emph{Case 1: $|\lambda|<1$.} Identify $f$ with its local coordinate expression $\widehat f$. Let $0$ be the fixed point. We have $f(z)=\lambda z+O(z^2)$ in $\mathbb D_{r_0}$ for some $r_0<1$, and
\[
|f(z)|\le|\lambda z|+C|z|^2\le(|\lambda|+C|z|)|z|\qquad(|z|<r_0).
\]
Choose $c$ with $|\lambda|<c<\sqrt{|\lambda|}$, and then choose $0<r\le r_0$ so that $|\lambda|+Cr\le c$. Then $|f(z)|\le c|z|$ for every $z\in\mathbb D_r$, so $f(z)\in\mathbb D_r$ and
$|f^{\circ n}(z)|\le c^n|z|<c^nr\to0$ as $n\to\infty$.
% Source page 4
This is precisely the push/pull we were discussing. There is $r<1$ such that $|f(z)|<c|z|$ on $\mathbb D_r$, with $c<1$. Let $\{z_n\}_n$ be the orbit of $z_0\in\mathbb D_r$. All these orbits converge to the origin, since $|z_n|\le rc^n$, and $|z_{n+1}-\lambda z_n|\le C|z_n|^2\le Cr^2c^{2n}$. Set $w_n=z_n/\lambda^n$. Then
\begin{align*}
|w_{n+1}-w_n|
&=\left|\frac{z_{n+1}}{\lambda^{n+1}}-\frac{z_n}{\lambda^n}\right|
=\frac{|z_{n+1}-\lambda z_n|}{|\lambda|^{n+1}}\\
&\le\frac{Cr^2c^{2n}}{|\lambda|^{n+1}}
=\frac{Cr^2}{|\lambda|}\left(\frac{c^2}{|\lambda|}\right)^n.
\end{align*}
The sequence $\{w_n\}_n$ is Cauchy, and its geometric bound is independent of $z_0$. The map $\phi_n(z_0)=f^{\circ n}(z_0)/\lambda^n$ is holomorphic on $\mathbb D_r$, with $\phi_n(f(z_0))=\lambda\phi_{n+1}(z_0)$. Taking $\phi=\lim_{n\to\infty}\phi_n$ gives $\phi\circ f=\lambda\phi$. Also, $\phi_n'(0)=1$, so $\phi'(0)=1$, and $\phi$ is a local conformal isomorphism around $0$.

\emph{Case 2: $|\lambda|>1$.} Since $f'(0)\ne0$, $f$ is locally invertible. Its local inverse has multiplier $0<|1/\lambda|<1$ at $0$.
% Source page 5
For uniqueness, if $\phi,\psi$ are two such Koenigs maps, then $\psi\circ\phi^{-1}$ commutes with $z\mapsto\lambda z$. If $\psi\circ\phi^{-1}(w)=\sum_{n\ge1}b_nw^n$, then $\lambda b_n=b_n\lambda^n$. Thus $b_k=0$ for every $k\ge2$, and $\psi=b_1\phi$.
\end{proof}
\subsection{Superattracting fixed points}
If $f'(0)=0$, then locally $f(z)=a_nz^n+a_{n+1}z^{n+1}+\cdots$. Thus $f$ has a zero of order $n$ at $0$, and in a suitable domain $f(z)=h(z)^nz^n$. Close to the origin in this domain, $f$ acts almost like the polynomial $a_nz^n$. The following theorem makes a much sharper claim.
\setcounter{theorem}{11}\begin{theorem}[B\"ottcher]\label{cd2:bottcher}
If $0$ is a superattracting fixed point of $f$, with a zero of order $n$ at $0$, there is a map $\phi$, unique up to multiplication by an $(n-1)$th root of unity, such that the following diagram commutes.
\begin{figure}[H]\centering
\includegraphics[width=.32\textwidth]{cd2-fig-03.pdf}
\FigureTag{CD2.3}{fig:cd2-03}
\end{figure}
\end{theorem}
% Source page 6
\begin{proof}
Locally around $0$, write
\[
f(z)=b_0z^n+b_1z^{n+1}+b_2z^{n+2}+\cdots.
\]
Since $cf(z/c)=b_0z^n/c^{n-1}+\cdots$, choosing $c^{n-1}=b_0$ allows us to assume that the leading coefficient is $1$.

Choose $r$ small enough that $f$ maps $\mathbb D_r$ into itself and
\[
f(z)=z^n(1+\eta(z)),\qquad \eta(0)=0,
\]
with $|\eta(z)|<1/2$ on $\mathbb D_r$. Put
\[
\mathbb H_r=\{Z\in\mathbb C:\Re Z<\log r\}.
\]
The exponential map covers $\mathbb D_r\setminus\{0\}$, and $f$ has a lift
\[
F(Z)=\log f(e^Z)=nZ+\log(1+\eta(e^Z))
\]
on $\mathbb H_r$, where the branch is chosen so that $F(Z)-nZ\to0$ as $\Re Z\to-\infty$. After decreasing $r$ if necessary,
\[
|F(Z)-nZ|<1\qquad(Z\in\mathbb H_r).
\]

Set
\[
\Phi_k(Z)=\frac{F^{\circ k}(Z)}{n^k}.
\]
Then
\[
|\Phi_{k+1}(Z)-\Phi_k(Z)|
=\frac{|F(F^{\circ k}(Z))-nF^{\circ k}(Z)|}{n^{k+1}}
<\frac1{n^{k+1}}.
\]
Thus $\{\Phi_k\}_k$ converges locally uniformly to a holomorphic map $\Phi$, and
\[
\Phi\circ F=n\Phi.
\]
Moreover, $F(Z+2\pi i)=F(Z)+2\pi in$, so
\[
\Phi_k(Z+2\pi i)=\Phi_k(Z)+2\pi i
\]
for every $k$, and the same identity holds for $\Phi$. Consequently
\[
\phi(e^Z)=e^{\Phi(Z)}
\]
is well-defined on $\mathbb D_r\setminus\{0\}$. Since $\Phi(Z)-Z\to0$ as $\Re Z\to-\infty$, the map extends holomorphically across $0$, with $\phi(0)=0$ and $\phi'(0)=1$. Finally,
\[
\phi(f(e^Z))=e^{\Phi(F(Z))}=e^{n\Phi(Z)}=\phi(e^Z)^n.
\]
Hence $\phi\circ f=(z\mapsto z^n)\circ\phi$, and $\phi$ is a local conformal isomorphism at $0$.

Uniqueness again follows because $\phi\circ\psi^{-1}$ commutes with $z\mapsto z^n$. Comparing its first nonzero Taylor coefficient beyond the linear term shows that it is multiplication by an $(n-1)$th root of unity.
\end{proof}
\begin{remark}
A crucial fact is that $f(z)\ne0$ for $z\in\mathbb D_r\setminus\{0\}$, so the lift $\log f(e^Z)$ makes sense. This is a useful technique for bounding quantities: move the problem to another hyperbolic surface and use the estimates supplied by Pick's theorem.
\end{remark}
% Source page 8
How far can these local descriptions be extended? We now pass from local to global linearisation, focusing on rational maps on $\widehat{\mathbb C}$.
\subsection{Global linearisation}
Koenigs' function $\phi$ extends holomorphically to the basin of attraction $\mathcal A$ of the fixed point $\widehat p$.
\begin{figure}[H]\centering
\includegraphics[width=.32\textwidth]{cd2-fig-04.pdf}
\FigureTag{CD2.4}{fig:cd2-04}
\end{figure}
\begin{proof}
Let $U_0$ be a forward-invariant neighbourhood of $\widehat p$ on which the local Koenigs coordinate $\phi$ is defined and satisfies $\phi\circ f=\lambda\phi$. For every $z\in\mathcal A$, there is a neighbourhood $U$ of $z$ and an integer $N$ such that $f^{\circ N}(U)\subset U_0$. Define
\[
\Phi(w)=\frac{\phi(f^{\circ N}(w))}{\lambda^N}\qquad(w\in U).
\]
If $M\ge N$ also satisfies $f^{\circ M}(U)\subset U_0$, then
\[
\frac{\phi(f^{\circ M}(w))}{\lambda^M}
=\frac{\lambda^{M-N}\phi(f^{\circ N}(w))}{\lambda^M}
=\frac{\phi(f^{\circ N}(w))}{\lambda^N}.
\]
Thus the definition is independent of $N$. The local definitions agree on overlaps and patch to a holomorphic map $\Phi\colon\mathcal A\to\mathbb C$ that extends $\phi$ and satisfies $\Phi\circ f=\lambda\Phi$.
\end{proof}
The local inverse $\psi=\phi^{-1}$ of Koenigs' function extends to a meromorphic function on $\mathbb{C}$ when $\widehat p$ is a repelling fixed point.
\begin{figure}[H]\centering
\includegraphics[width=.36\textwidth]{cd2-fig-05.pdf}
\FigureTag{CD2.5}{fig:cd2-05}
\end{figure}
The proof follows the same pattern: choose $N$ so that $w/\lambda^N\in\mathbb D_\varepsilon$, apply $\phi^{-1}$, and compose with $f^{\circ N}$. Locally,
\[
\psi(w)=f^{\circ N}\left(\phi^{-1}\left(\frac w{\lambda^N}\right)\right),\qquad
f\circ\psi(w)=f^{\circ(N+1)}\left(\phi^{-1}\left(\frac w{\lambda^N}\right)\right),
\]
while $\psi\circ\lambda(w)=f^{\circ N}\bigl(\phi^{-1}(w/\lambda^{N-1})\bigr)$.
\begin{remark}
Such an extension is not possible for B\"ottcher coordinates: to make the diagram commute one must deal with $p\mapsto\sqrt[n^k]{\phi(f^{\circ k}(p))}$, but the root is not well-defined. One can, however, extend $|\phi|$, since roots in $\mathbb R_+$ are well-defined.
\end{remark}

\subsection{The phenomenon of critical points: analytic aspects}
Critical points govern rational dynamics in three significant ways:
\begin{enumerate}
\setcounter{enumi}{0}\item Critical points form the boundary beyond which Koenigs and B\"ottcher coordinates cannot be extended biholomorphically (analytic).
\setcounter{enumi}{1}\item Critical points determine the number of attracting periodic points or orbits (analytic).
\setcounter{enumi}{2}\item They determine the topological shape of the Julia set (topological).
\end{enumerate}
The analysis of a rational map is essentially determined by its roots and poles, whereas in dynamics its critical points become essential. We address these phenomena one by one. Unless otherwise stated, $f$ is a rational map with $\deg(f)\ge2$, and the surface is $\widehat{\mathbb C}$.
\subsubsection{Biholomorphic extension}
\setcounter{definition}{7}\begin{definition}
Given an attracting fixed point $\widehat p$ of a map $f\colon S\to S$, let $\mathcal A_0$ denote the connected component of its basin of attraction that contains $\widehat p$.
\end{definition}
\setcounter{theorem}{12}\begin{theorem}\label{cd2:maximal}
There is a maximal $r$ such that $\psi\colon\mathbb D_r\to\mathcal A_0$ is a holomorphic inverse of the Koenigs coordinate $\phi$. The closed disc $\overline{\mathbb D_r}$ and $\psi(\overline{\mathbb D_r})$ are homeomorphic, and $\psi(\partial\mathbb D_r)$ contains a critical point of $f$.
\end{theorem}
% Source page 11
Thus critical points obstruct biholomorphic extension within $\mathcal A$.
% Source page 13
\setcounter{theorem}{13}\begin{theorem}\label{cd2:maximal-bottcher}
There is a maximal $0<r\le1$ such that the B\"ottcher coordinate $\phi$ has a biholomorphic inverse on $\mathbb D_r$. If $r=1$, then $\mathbb D\cong\mathcal A_0$, and the superattracting point $\widehat p$ is the only critical point in $\mathcal A_0$. Otherwise there is another critical point in $\mathcal A_0$, on $\partial\psi(\mathbb D_r)$.
\end{theorem}
\begin{remark}
We defer the homeomorphism part to \BlogPost{parabolic-fixed-points-and-the-flower-theorem}{the next post}, where we explore topological properties, including continuous extension to $\partial\mathbb D_r$ of the inverses of B\"ottcher and Koenigs coordinates, and much more.
\end{remark}
\subsubsection{The number of attracting periodic points}
% Source page 15
\setcounter{theorem}{14}\begin{theorem}\label{cd2:finite-attractors}
There are finitely many attracting periodic orbits.
\end{theorem}
\begin{proof}
For every attracting periodic orbit, the union of its immediate-basin components contains a critical point. Distinct attracting cycles have disjoint immediate basins, while a rational map has only finitely many critical points.
\end{proof}
Critical points therefore bound the number of attracting periodic orbits. Let us try to extend this theorem to noncompact Riemann surfaces and nonrational maps.
\setcounter{definition}{8}\begin{definition}
For $f\colon S\to S'$, a \emph{critical value} is the image under $f$ of a critical point.
\end{definition}
\setcounter{definition}{9}\begin{definition}
If there is a curve $\gamma\colon[0,1)\to S$ that eventually leaves compact sets, but $\lim_{t\to1}f(\gamma(t))=v$, then $v$ is an \emph{asymptotic value}.
\end{definition}
% Source page 16
\setcounter{theorem}{15}\begin{theorem}
Every proper holomorphic self-map $f\colon S\to S$ has a critical value in every immediate basin of an attracting orbit.
\end{theorem}
