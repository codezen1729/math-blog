\section*{Polynomial-Like Maps and the Straightening Theorem}
% BLOG-CONTENT-BEGIN
\subsection{Cut and Paste Surgery}
% Source page 1
Given $f\colon S\to S$, the following are the analytic parameters of the
dynamics of $f$ on $S$, locally speaking. In almost all our treatments,
$S=\mathbb C$ or $\widehat{\mathbb C}$, $f$ is rational, and $\deg(f)\ge2$.
\begin{itemize}
 \item Multipliers of fixed points.
 \item Number of attracting and repelling cycles.
 \item Nature of the Fatou components.
 \item Rotation numbers of Siegel discs and Herman rings.
 \item Rate of expansion, if $f$ is hyperbolic.
 \item Holomorphic degree of $f$, if $f$ is a branched covering.
\end{itemize}
\emph{Cut-and-paste surgery} alters these parameters by gluing maps with
different dynamical data along selected subdomains. The Straightening
Theorem is a central example.

\setcounter{definition}{26}\begin{definition}[Polynomial-like map]
Let $U,V$ be bounded simply connected domains of $\mathbb C$ such that
$\partial U,\partial V$ are analytic curves and
$\overline U\subset V\subset\mathbb C$.
A map $f\colon U\to V$ is said to be \emph{polynomial-like} if it is a
proper branched holomorphic covering of degree $d$.
\end{definition}

% Source page 3
\begin{remark}[1]
With this definition we are trying to study $f$ locally around its
critical points, where it was observed to have polynomial-like behaviour.
Polynomials are polynomial-like, so this definition genuinely generalises
them.
\end{remark}
Here $K(f):=\bigcap_{n\ge0}f^{-n}(V)$ is the set of points that never escape under the polynomial-like map.
\setcounter{theorem}{51}\begin{theorem}[Straightening Theorem]
\label{thm:cd13-straightening}
Every polynomial-like map $f\colon U\to V$ of degree $d$ is hybrid equivalent to a polynomial $P$ of degree $d$: there is a quasiconformal conjugacy whose Beltrami coefficient vanishes almost everywhere on $K(f)$ \cite{douadyhubbard1985polynomiallike}.
\end{theorem}
\paragraph{Note.}
In other words, the ellipse field of a polynomial-like map can be
straightened into a circle field, hence the name.
The proof glues the polynomial-like map to an external polynomial model,
then straightens the resulting invariant ellipse field.
% Source page 4 (continuation of the previous sentence).

\begin{proof}
Let $\rho>1$, and let
\[
 R\colon\widehat{\mathbb C}\setminus\overline V
 \longrightarrow\widehat{\mathbb C}\setminus\overline{\mathbb D}_{\rho^d}
\]
be a Riemann map fixing $\infty$. Since $\partial V$ is analytic, $R$
extends analytically to its boundary $\partial V$.
Call such an extension $\psi_1\colon\partial V\to S^1_{\rho^d}$.
Let $\widehat f=f|_{\partial U}$ and
$\widetilde f=\psi_1\circ\widehat f\colon\partial U\to S^1_{\rho^d}$.
The map $\widehat f$ is an analytic covering of degree $d$, while
$z\mapsto z^d$ is a covering from $S^1_\rho$ to $S^1_{\rho^d}$ of the
same degree. Hence $\widetilde f$ lifts through $z\mapsto z^d$ to an
analytic orientation-preserving diffeomorphism
$\psi_2\colon\partial U\to S^1_\rho$, giving
$\psi_1(f(z))=\psi_2(z)^d$ for every $z\in\partial U$.
\begin{figure}[H]
 \centering\includegraphics[width=.35\textwidth]{cd13-fig-01.pdf}
 \FigureTag{CD13.1}{fig:cd13-01}
\end{figure}
Let
\[
 A_0=V\setminus\overline U,\qquad
 A_{\rho,\rho^d}:=\{\zeta:\rho<|\zeta|<\rho^d\}.
\]
The maps $\psi_1,\psi_2$ are analytic on $\partial V,\partial U$,
so there exists a quasiconformal map
$\psi\colon A_0\to A_{\rho,\rho^d}$ which extends continuously to the
closures and satisfies
\[
 \psi|_{\partial V}=\psi_1,\qquad \psi|_{\partial U}=\psi_2.
\]
Now it is time to perform the surgery.

% Source page 5
On the annulus $A_0$, use the quasiconformal map $\psi$ to glue
$f$ to the external holomorphic model. The resulting map supplies the
invariant ellipse field that will be straightened to a circle field.
Define $F\colon\widehat{\mathbb C}\to\widehat{\mathbb C}$, the result of surgery on $f$, by
\[
 F(z):=
 \begin{cases}
  f(z),&z\in U,\\
  R^{-1}\bigl(\psi(z)^d\bigr),&z\in A_0,\\
  R^{-1}\bigl(R(z)^d\bigr),&z\in\widehat{\mathbb C}\setminus\overline V.
 \end{cases}
\]
On the two boundary curves these formulas have the same continuous
extensions. Note that $F$ is a quasiregular branched covering of degree
$d$. If we find an $F$-invariant Beltrami form, the Mapping Theorem will
conclude the proof.

Let $\mu_0$ be the circle field and let $\mu_\psi=\psi^*\mu_0$. Put
$E_0=A_0$ and, for $n\geq1$, put
\[
 E_n=\{z\in U:f^j(z)\in U\ (0\leq j<n),\ f^n(z)\in A_0\}.
\]
Define the Beltrami form $\mu$ by
\[
\mu(z):=
\begin{cases}
  (f^n)^*\mu_\psi,&z\in E_n,\quad n\geq0,\\
  \mu_0,&z\in K(f)\cup(\widehat{\mathbb C}\setminus V).
\end{cases}
\]
On the remaining preimages of the boundary curves, which have area zero, set $\mu=\mu_0$.
Since holomorphic pullback preserves eccentricity,
$\|\mu\|_\infty=\|\mu_\psi\|_\infty<1$, and the construction gives
$F^*\mu=\mu$. Hence there exists a quasiconformal map
$\phi\colon\widehat{\mathbb C}\to\widehat{\mathbb C}$, normalized by
$\phi(\infty)=\infty$, such that
\[
 P:=\phi\circ F\circ\phi^{-1}
\]
is holomorphic on $\widehat{\mathbb C}$. It has degree $d$, and
$P^{-1}(\infty)=\{\infty\}$, so $P$ is a polynomial of degree $d$.
Since $F=f$ on $U$, the map $\phi$ conjugates $f$ to $P$ on a
neighbourhood of $K(f)$. Moreover, $\mu=\mu_0$ on $K(f)$, so the
Beltrami coefficient of $\phi$ vanishes almost everywhere there.
Thus $f$ and $P$ are hybrid equivalent.
\end{proof}
