\section*{Transformations: Rotating, Reflecting, and Cutting Figures}

\subsection{Transformations}
\setcounter{theorem}{91}\begin{lemma}[Transformation techniques]\label{lemma:lb1-94}
We can rotate, reflect, and invert figures. Transformation techniques are powerful tools, as the following examples show.
\end{lemma}
\paragraph{Transformation I.}
$AB=BC=CA$, $PA=3$, $PB=4$, and $PC=5$. Find $[ABC]$.
\begin{figure}[H]\centering
\includegraphics[width=.68\linewidth]{lb1-fig-34.pdf}
\FigureTag{LB1.34}{fig:lb1-34}
\end{figure}
Rotate $\triangle APB$ by $60^\circ$ to $AP'C$.
Similarly obtain $BP''C$ and $BP'''A$.
Now $[APB]=[AP'C]$, $[BPC]=[BP'''A]$, and $[APC]=[BP''C]$.
Thus $[\text{hexagon}]=2[ABC]$.
That hexagon consists of three quadrilaterals, each consisting of a $3$--$4$--$5$ right triangle and an equilateral triangle.
Hence
\[[\text{hexagon}]=3\left(\frac12\cdot3\cdot4\right)
+\frac{\sqrt3}{4}(3^2+4^2+5^2),
\qquad[ABC]=9+\frac{25\sqrt3}{4}.\]
\paragraph{Transformation II: Pompeiu's theorem.}
Given a point $P$ inside an equilateral triangle $ABC$, the lengths $PA,PB,PC$ form the sides of a triangle.
Rotate $\triangle ABC$ through $60^\circ$ about $C$ to $A'AC$.
Then $\triangle PP'C$ is equilateral, so $PP'=PC$.
The triangle $APP'$ has side lengths $PA,PB,PC$.
\begin{figure}[H]\centering
\includegraphics[width=.70\linewidth]{lb1-fig-35.pdf}
\FigureTag{LB1.35}{fig:lb1-35}
\end{figure}
There is another beautiful extension: if $P$ lies on arc $BC$ of the circumcircle of the equilateral triangle $ABC$, then $PA=PB+PC$.

\paragraph{Transformation III.}
Choose one point on each side of a unit square.
Let the quadrilateral formed by those points have sides $a,b,c,d$.
Prove that $a+b+c+d\geq2\sqrt2$.
\begin{figure}[H]\centering
\includegraphics[width=.42\linewidth]{lb1-fig-36.pdf}
\FigureTag{LB1.36}{fig:lb1-36}
\end{figure}
Use multiple reflections in the sides to obtain a solution.
A reflection argument gives the following solution.
The perimeter of $ABCD$ is minimal if it is a closed light path.
All those light polygons have perimeter $2\sqrt2$: when you join the midpoints of the sides, you get a light polygon with perimeter $2\sqrt2$.
Therefore $a+b+c+d\geq2\sqrt2$. Q.E.D.!

\paragraph{Transformation IV.}
Let us solve the IMO 1996 cut-and-paste problem using inversion.
\begin{figure}[H]\centering
\includegraphics[width=.69\linewidth]{lb1-fig-37.pdf}
\FigureTag{LB1.37}{fig:lb1-37}
\end{figure}
Invert the inner triangle with respect to $I(P,r)$.
The angle condition on $P$ implies $\alpha+\theta=\beta+\gamma$, so $\triangle A'B'C'$ is isosceles.
Thus $A'B'=A'C'$.
Using the distance relation,
\[\frac{AB\cdot r^2}{PA\cdot PB}=\frac{AC\cdot r^2}{PC\cdot PA}
\quad\Longrightarrow\quad\frac{AB}{BP}=\frac{AC}{PC}.\]
This proves the claimed ratio.

The cut-and-paste construction above has another application.
\paragraph{Tournament of Towns (2002).}
In triangle $ABC$, a point $P$ satisfies $\angle ABP=\angle ACP$ and $\angle PBC=\angle PAC$.
Prove using pure geometry that $P$ is the orthocentre.
\begin{figure}[H]\centering
\includegraphics[width=.65\linewidth]{lb2-fig-12.pdf}
\FigureTag{LB2.12}{fig:lb2-12}
\end{figure}
Trigonometric Ceva gives a quick proof, but the following argument is purely geometric.
% Source page 13 - right
\begin{figure}[H]\centering
\includegraphics[width=.92\linewidth]{lb2-fig-13.pdf}
\FigureTag{LB2.13}{fig:lb2-13}
\end{figure}
Rearranging the pieces reveals the relevant angles.
The rearranged pieces give $\angle PAB=\angle PCB=z$, and then
$2(x+y+z)=180^\circ$.
Thus $x+y+z=90^\circ$, so $P$ is the orthocentre of $ABC$.
