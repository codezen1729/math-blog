\section{Normalization and the Mandelbrot set}
Start with $f(z)=az^2+bz+c$. Conjugating by a dilation gives
\[
\lambda^{-1}f(\lambda z)=z^2+\alpha z+\beta,
\qquad \lambda=\frac1a.
\]
We then conjugate by a translation:
\[
\begin{aligned}
(z-d)\circ(z^2+\alpha z+\beta)\circ(z+d)
 &=(z+d)^2+\alpha(z+d)+\beta-d\\
 &=z^2+(2d+\alpha)z+d^2-d+\beta.
\end{aligned}
\]
Taking $d=-\alpha/2$, the critical point, gives the normal form $z^2+c$, denoted by $P_c$.

Now suppose that $\varphi$ is a M\"obius map, $c\ne c'$, and
\[
\varphi^{-1}\circ(z^2+c)\circ\varphi=z^2+c'.
\]
On $\Chat$, the points $\{\infty,0\}$ are critical points of both $z^2+c$ and $z^2+c'$. Hence $\varphi\equiv\lambda z+d$, and there are unique $\lambda,d$ conjugating any quadratic polynomial to $z^2+c$. Thus
\[
\Moduli(\text{quadratic polynomials})\cong\C.
\]

\setcounter{definition}{0}\begin{definition}[Mandelbrot set]
The \emph{Mandelbrot set} is
\[
\M:=\{c\in\C\mid (P_c^{\circ n}(0))_{n\ge0}\text{ is bounded}\}.
\]
\end{definition}

\subsection{Critical points and connectedness}
\setcounter{theorem}{0}\begin{theorem}[Connected and Cantor filled Julia sets]\label{thm:critical-connectedness}
Let $P$ be any polynomial, and let $K_P$ be its filled Julia set. Then:
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $\Crit(P)\subset K_P$ if and only if $K_P$ is connected.
\setcounter{enumi}{1}\item If $\Crit(P)\cap K_P=\varnothing$, then $K_P$ is a Cantor set.
\end{enumerate}
\end{theorem}

\begin{proof}
Write $P(z)=a_dz^d+a_{d-1}z^{d-1}+\cdots+a_0$, and set
\[
A=\max_{0\le i\le d}\{|a_i|\},
\qquad R=\max\left\{1,\frac{2+A}{|a_d|}\right\}.
\]
Then
\[
\begin{aligned}
|P(z)|&\ge |a_d||z|^d-\bigl(|a_{d-1}||z|^{d-1}+\cdots+|a_0|\bigr)\\
 &\ge |a_d||z|^d-A|z|^{d-1}\\
 &\ge |z|^{d-1}(|a_d||z|-A)\ge2|z|
 \qquad (|z|\gg R).
\end{aligned}
\]
Thus there is an $R_0$ such that, if $z\in(\overline{D_R(0)})^c$, then $P^{\circ n}(z)\to\infty$. Consequently,
\begin{equation}\label{eq:filled-intersection}
K_P=\bigcap_{n=1}^{\infty}P^{-n}(\overline{D_R(0)}).
\tag{$*$}
\end{equation}
Set $U_0=\Chat\setminus\overline{D_R(0)}$ and $U_n=P^{-1}(U_{n-1})$. We have $U_0\subset U_1\subset U_2\subset\cdots$ and, by \eqref{eq:filled-intersection}, $\bigcup_{n\ge0}U_n=\C\setminus K_P$.

\textit{(i)} If $\Crit(P)\subset K_P$, then $P\colon U_n\to U_{n-1}$ is a branched cover ramified at $\{\infty\}$, with degree $d$. Hence $U_n$ is connected. Since $\chi(U_0)=1$, the Riemann--Hurwitz formula gives
\[
\chi(U_n)=d\chi(U_{n-1})-(d-1)=1
\]
by induction. Therefore $U_n$ is a hyperbolic Riemann surface with $U_n\cong\D$.

% Source page 3
If $\Crit(P)\not\subset K_P$, then $P\colon U_n\to U_{n-1}$ has other points where it is ramified. Thus $\chi(U_n)<1$, so $U_n$ is not simply connected. Its complement $U_n^c$ has more than two components, and $K_P$ is not connected.

\textit{(ii)} If $\Crit(P)\cap K_P=\varnothing$, then \eqref{eq:filled-intersection} again gives an $m$ large enough that
\[
\Crit(P)\cap P^{-m}(D_R(0))=\varnothing.
\]
Put $V=P^{-m}(D_R)$ and $W=P^{-(m-1)}(D_R)$. Then $\overline V\subset W$ and $P\colon V\to W$ is a covering map. Let $(z,\xi)$ be a tangent vector at $z$. By Pick's theorem applied to $V\hookrightarrow W$,
\[
|(z,\xi)|_V\ge C|(z,\xi)|_W\qquad(C>1),
\]
and
\[
|(P(z),P'(z)\xi)|_W=|(z,\xi)|_V\ge C|(z,\xi)|_W.
\]
Let $M=\max\{\diam(\text{components of }V)\}$. The maximum diameter of a component of $P^{-n}(V)$ is $MC^{-n}$, which tends to zero as $n\to\infty$. Hence $K_P=\bigcap_{n=0}^{\infty}P^{-n}(V)$ is a Cantor set.
\end{proof}

\setcounter{theorem}{1}\begin{corollary}[Connected Julia sets]
\[
\M=\{c\mid K(P_c)\text{ is connected}\}
    =\{c\mid J(P_c)\text{ is connected}\}.
\]
\end{corollary}

\subsection{Escape bounds and the real slice}
\setcounter{theorem}{2}\begin{theorem}[Basic properties of the Mandelbrot set]\label{thm:basic-properties}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item $\M\subseteq\{c\mid |c|\le2\}$, and $\M$ is closed and simply connected.
\setcounter{enumi}{1}\item $\M\cap\mathbb R=[-2,1/4]$.
\setcounter{enumi}{2}\item $|P_c^{\circ n}(0)|\le2$ for all $n\ge1$ if and only if $c\in\M$.
\end{enumerate}
\end{theorem}

\begin{proof}
\textit{(i)} If $|c|>2$, induction gives
\[
|P_c^{\circ n}(0)|\ge |c|(|c|-1)^{2^{n-1}},\qquad n\ge1.
\]
Thus $|P_c^{\circ n}(0)|\to\infty$ and $c\notin\M$, giving (i).

\textit{(iii)} Suppose $|P_c^{\circ m}(0)|=2+\delta>2$ for some $m\ge1$. If $|c|=|P_c(0)|>2$, then $c\notin\M$. If $|c|\le2$, then
\[
|P_c^{\circ(m+1)}(0)|\ge(2+\delta)^2-2\ge2+4\delta.
\]
By induction, $|P_c^{\circ(m+k)}(0)|\ge2+4^k\delta\to\infty$, so again $c\notin\M$. If $c_m\to c$ with $c_m\in\M$, then $|P_c^{\circ n}(0)|\le2$ for every $n$; hence $\M$ is closed.

\emph{Aside.} If $\C\setminus\M$ has bounded components, then $|P_c^{\circ n}(0)|>2$ for some $n$, and the maximum modulus principle disallows this.

\textit{(ii)} For $c\in\mathbb R$, the polynomial $P_c(x)-x$ has no real roots if $c>1/4$, one real root if $c=1/4$, and two real roots if $c<1/4$. If $c>1/4$, then $P_c^{\circ n}(0)\to\infty$; otherwise the orbit would accumulate at an $x\in\mathbb R$ with $P_c(x)=x$.

If $c\le1/4$, let $a=(1+\sqrt{1-4c})/2$ be the larger of the two roots of $P_c(x)-x$. If also $c\ge-2$, then $a\ge|c|=|P_c(0)|$.

% Source page 5
Therefore
\[
|P_c^{\circ n}(0)|\le a
\quad\Longrightarrow\quad
|P_c^{\circ(n+1)}(0)|
=|(P_c^{\circ n}(0))^2+c|\le a^2+c=a.
\]
The sequence $\{P_c^{\circ n}(0)\}_n$ is bounded, and $\M\cap\mathbb R=[-2,1/4]$.
\end{proof}

\subsection{Attracting cycles and stability}
\setcounter{theorem}{3}\begin{theorem}[Continuation of attracting cycles]\label{thm:attracting-continuation}
Let $a\in\Int(\M)$, and suppose $P_a$ has an attracting cycle of length $m$. Let $W$ be the component of $\Int(\M)$ containing $a$. Then, for every $c\in W$, $P_c$ has an attracting cycle $\{z_1(c),\ldots,z_m(c)\}$ of length $m$, and $c\mapsto z_i(c)$ is holomorphic for every $1\le i\le m$.
\end{theorem}

\begin{proof}
Consider $P_c^{\circ m}(z)-z$. By the implicit function theorem, we obtain an attracting periodic point of period $m$, say $z_1(c)$, in a small neighbourhood of $a$. The sequence $\{P_c^{\circ jm}(0)\}$ is bounded. Hence all subsequences of this sequence converge to an analytic continuation of $z_1(c)$ over $W$.

The point $z_1(c)$ could be repelling, but convergence $P_c^{\circ jm}(0)\to z_1(c)$ would then imply $P_c^{\circ jm}(0)=z_1(c)$, by the local Koenigs dynamics. This happens for countably many points (false by the implicit function theorem).

% Source page 6
Thus $|(P_c^{\circ m})'(z_1(c))|\le1$. By the maximum modulus principle, $z_1(c)$ is an attracting fixed point on all of $W$. The other functions are $\{P_c^{\circ i}(z_1(c))\}_{0\le i<m}$.
\end{proof}

\setcounter{theorem}{4}\begin{theorem}[Density of the interior]\label{thm:interior-dense}
$\Int(\M)$ is dense in $\M$.
\end{theorem}
\begin{proof}
Let $U$ be a disk that meets $\partial\M$, with $0\notin U$.
\begin{figure}[H]
\centering
\includegraphics[width=7.5cm,height=4.7cm,keepaspectratio]{qf-fig-01.pdf}
\FigureTag{QF1}{fig:qf-01}
\end{figure}
Suppose $U$ has no $c$ for which $0$ is periodic under $P_c$. Consider a branch of $\sqrt{-c}$ on $U$. Then $P_c^{\circ n}(0)\ne\sqrt c$, because $P_c^{\circ(n+1)}(0)=0$. Consequently, the family
\[
\left\{\frac{P_c^{\circ n}(0)}{\sqrt{-c}}\right\}_n
\]
omits $\{0,1,\infty\}$ and is normal in $c$. But $U$ meets $\partial\M$, so for some $c\in U$ the sequence explodes, whereas for some $c$ it remains bounded, violating normality.
\end{proof}

\setcounter{theorem}{5}\begin{corollary}[$J$-stability]
\[
\{c\mid P_c\text{ is }J\text{-stable}\}=\C\setminus\partial\M.
\]
\end{corollary}
\begin{proof}
By definition, if $c\in\Int(\M)$, the sequence $\{P_c^{\circ n}(0)\}_n$ is bounded and hence forms a normal family, so $c$ is $J$-stable. If $c\in\Ext(\M)$, then $P_c^{\circ n}(0)\to\infty$, again giving normality and $J$-stability. As in the preceding theorem, normality fails exactly on the boundary.
\end{proof}

\section{Components, invariant line fields, and corrections}
\subsection{Summary}
\begin{enumerate}
\setcounter{enumi}{0}\item $\M$ is closed and simply connected, and is contained in $\{c\mid |c|\le2\}$.
\setcounter{enumi}{1}\item $\Int(\M)$ is dense in $\M$.
\setcounter{enumi}{2}\item $\{c\mid P_c\text{ is }J\text{-stable}\}=\C\setminus\partial\M$.
\setcounter{enumi}{3}\item Connected components of $\M$ are characterized by the length of the attracting cycle they admit, if they do admit one.
\setcounter{enumi}{4}\item $\M=\{c\mid J(P_c)\text{ is connected}\}$.
\end{enumerate}

\subsection{Non-hyperbolic components}
\setcounter{theorem}{6}\begin{theorem}[Invariant line fields]\label{thm:line-fields}
If $c\in\M$, then $c$ belongs to a non-hyperbolic component of $\M$ if and only if $J(P_c)$ has positive measure and carries an invariant line field.
\end{theorem}
\begin{proof}
If $c$ belongs to a non-hyperbolic component, then $P_c$ has neither an attracting cycle nor an indifferent cycle, because indifferent points are persistently indifferent. A polynomial has no Herman ring, so by the Sullivan--Fatou classification, $J(P_c)$ is full.

\FigureTag{QF2}{fig:qf-02}
\end{minipage}\hfill
\begin{minipage}[c]{.68\linewidth}
\centering
\includegraphics[width=\linewidth,height=4.8cm,keepaspectratio]{qf-fig-03.pdf}
\FigureTag{QF3}{fig:qf-03}
\end{minipage}
\end{figure}
The critical points of $P_\lambda$ are in $K(P_\lambda)$, so the B\"ottcher coordinate extends conformally. Hence $\phi_\lambda(z)$ is a holomorphic motion of $\C\setminus K(P_c)$.

% Source page 8
By the $\lambda$-lemma, $\phi_\lambda$ extends quasiconformally to $\overline\phi_\lambda\colon\Chat\to\Chat$.
\begin{figure}[H]
\centering
\includegraphics[width=12cm,height=5.1cm,keepaspectratio]{qf-fig-04.pdf}
\FigureTag{QF4}{fig:qf-04}

If $J(P_c)$ has measure zero, then $P_c$ and $P_\lambda$ are conjugate by a M\"obius map. For $c\ne\lambda$, this is not possible.

Now suppose $J(P_c)$ has positive measure and supports an invariant line field $\mu$. Positive measure implies that $c\notin\Ext(\M)$, so $c\in\M$, as $P_c$ is not hyperbolic. Solve
\[
\frac{\partial_{\overline z}\phi_t}{\partial_z\phi_t}=t\mu
\]
to obtain a holomorphic family of quasiconformal maps $\phi_t\colon\C\to\C$, with $t\in\D$. Set $g_t=\phi_t\circ P_c\circ\phi_t^{-1}$, and normalize suitably to obtain $g_t(z)=z^2+c(t)$.

We prove that $c(t)$ is injective. If $c(t_1)=c(t_2)$ and $\phi_{t_1},\phi_{t_2}$ are conformal outside $\C\setminus J(P_c)$, then, by continuity, $\phi_{t_1}=\phi_{t_2}$ on $J(P_c)$. Hence $t_1\mu=t_2\mu$. By the open mapping theorem, $\{c(t)\mid t\in\D\}$ is an open set in $\M$ containing $c$.
\end{proof}

\subsection{MLC and a few corrections}
\paragraph{MLC.} $\partial\M$ is locally connected.

\textit{(i) Correction to Theorem~\ref{thm:attracting-continuation}.}
The point $z_1(c)$ could be repelling at uncountably many $c$. Then there is a $j$ such that $P_c^{\circ jm}(0)=z_1(c)$ for uncountably many $c$. By the identity theorem, $z_1(c)$ is a polynomial in $c$, and $P_c^{\circ m}(z_1(c))=z_1(c)$ can have only finitely many solutions. Hence $z_1(c)$ is repelling only at countably many points.

\textit{(ii) Indifferent points lie on $\partial\M$.}
Here the parameters in question are $\{c\mid P_c\text{ has an indifferent point}\}$. Consider the variety $P_c^{\circ m}(z)-z=0$ in $\C^2$, of dimension one. It has finitely many singular points. Away from these finitely many points, we have a Riemann surface $V$, and $V\to\C$, $(z,c)\mapsto c$, is a $2^m$-sheeted branched cover.

Consider an indifferent point in $\Int(\M)$, say $a$. Let $V_0$ be an irreducible component of $V$ containing $a$. The map
\[
V_0\longrightarrow\C,
\qquad (z,c)\longmapsto(P_c^{\circ m})'(z)
\]
would be locally constant around $a$, hence constant on $V_0$ (but $V_0$ projects onto $\C$).

\section{Quasicircles and uniformisation}
\subsection{The principal hyperbolic component}
\setcounter{theorem}{7}\begin{theorem}[Quasicircle Julia sets]
If $c$ is in the principal hyperbolic component, then $J(P_c)$ is a quasicircle.
\end{theorem}
\begin{proof}
This is true for a much more general class of functions: if $R$ is a hyperbolic rational map with two Fatou components, then $J(R)$ is a quasicircle.

Let $D_1,D_2$ be the two Fatou components. Choose $\gamma_1$ so that $U_1$ has no critical points, and let $\gamma_2=R^{-1}(\gamma_1)$. One can choose $\gamma_1,\gamma_2$ to be analytic, since $D_1$ is a holomorphic image of the disk (the extension of the Koenigs coordinate is simply connected).
The restriction $R|_{\gamma_2}\colon\gamma_2\to\gamma_1$ is a $d$-fold covering map, where $d=\deg(R)$.

Pick a Riemann map $\widetilde R\colon U_1^c\to\Chat\setminus\overline{\D}_{\rho^d}$. This extends analytically (conformally, and quasisymmetrically) to $\psi_1\colon\partial U_1\to S^1_{\rho^d}$.
\begin{figure}[H]
\centering
\begin{minipage}[c]{.45\linewidth}
\centering
\includegraphics[width=\linewidth,height=6.4cm,keepaspectratio]{qf-fig-05.pdf}
\FigureTag{QF5}{fig:qf-05}
\end{minipage}\hfill
\begin{minipage}[c]{.49\linewidth}
\centering
\includegraphics[width=\linewidth,height=4.7cm,keepaspectratio]{qf-fig-06.pdf}
\FigureTag{QF6}{fig:qf-06}
\end{minipage}
\end{figure}
We have $\psi_1(R(z))=\psi_2(z)^d$. Fill $U_1\setminus U_2$ with a quasiconformal map, via the Ahlfors--Beurling extension:
\[
\psi\colon\overline{U_1\setminus U_2}\longrightarrow\overline{A}_{\rho,\rho^d}.
\]
Define
\[
g_1(z)=
\begin{cases}
R(z),&z\in U_2,\\
\widetilde R^{-1}(\psi(z)),&z\in U_1\setminus U_2,\\
\widetilde R^{-1}(\widetilde R(z)^d),&z\in D_1\setminus U_1.
\end{cases}
\]

% Source page 11
Using $g_1$, we can produce a $g_1$-invariant Beltrami form on $D_1$:
\[
\mu_1=
\begin{cases}
0,&U_1^c,\\
\psi^*(0),&U_1\setminus U_2,\\
(f^{\circ n})^*\mu,&\text{elsewhere }
\left(\displaystyle\bigcup_{n=1}^{\infty}g_1^{-n}(U_1\setminus U_2)\right).
\end{cases}
\]
Similarly, produce $\mu_2$ for $D_2$, and set
\[
\mu=\begin{cases}\mu_i&\text{on }D_i,\\0&\text{elsewhere},\end{cases}
\qquad
g=\begin{cases}g_i&\text{on }D_i,\\0&\text{elsewhere}.\end{cases}
\]
By the measurable Riemann mapping theorem, there is a straightening map $\varphi$ such that $R_0:=\varphi\circ g\circ\varphi^{-1}$ is analytic. The map $R_0$ has critical fixed points of degree $d-1$ each at $0$ and $\infty$, since $g_1,g_2$ have the same. Thus $R_0\equiv z^d$, so $J(R)=\varphi(S^1)$, which is a quasicircle.
\end{proof}

\subsection{Uniformisation of the complement of the Mandelbrot set}
\setcounter{theorem}{8}\begin{theorem}[Uniformisation of $\C\setminus\M$]
Recall that, if $c\in\M^c$, then $A_c(\infty)$ is the only Fatou component, and the Julia set is totally disconnected.
\end{theorem}

% Source page 12
For the map $P_c$, we have $A_c(\infty)\subseteq\widetilde A_c(\infty)$, the basin of the superattracting point.
\begin{figure}[H]
\centering
\includegraphics[width=8.5cm,height=4.5cm,keepaspectratio]{qf-fig-07.pdf}
\FigureTag{QF7}{fig:qf-07}
\end{figure}
The B\"ottcher map satisfies
\[
\varphi_c(P_c(z))=\varphi_c(z)^2,
\qquad
\varphi_c(z)=\lim_{n\to\infty}[P_c^n(z)]^{2^{-n}}.
\]
Moreover,
\[
\begin{aligned}
\varphi_c(z)
 &=z\prod_{n=1}^{\infty}
 \left(\frac{P_c^n(z)}{P_c^{n-1}(z)^2}\right)^{2^{-n}}\\
 &=z\prod_{n=0}^{\infty}
 \left(1+\frac{c}{P_c^n(z)^2}\right)^{2^{-(n+1)}}.
\end{aligned}
\]
Hence, if defined,
\[
\varphi_c(c)=c\prod_{n=0}^{\infty}
 \left(1+\frac{c}{P_c^n(z)^2}\right)^{2^{-(n+1)}},
\]
which has a simple pole at $\infty$.

Using $\varphi_c(P_c(z))=\varphi_c(z)^2$, we can extend $\varphi_c$ conformally until we hit a critical point of $P_c$, namely $0$. Set $G_c(z)=\log|\varphi_c(z)|$. Then
\begin{equation}\label{eq:green-functional}
G_c(P_c(z))=2G_c(z).
\tag{$\star$}
\end{equation}
This can be extended to all of $\widetilde A_c(\infty)$.

For $G_c(z)>G_c(0)$, the map $\varphi_c(z)$ is analytic and maps that set conformally to $\{\zeta\mid |\zeta|>e^{G_c(0)}\}$. Also,
\[
K(P_c(z))=\bigcap_{n=1}^{\infty}
P_c^{-n}\bigl(\{z\mid G_c(z)\le G_c(0)\}\bigr).
\]
By \eqref{eq:green-functional}, as $z\to\partial\widetilde A_c(\infty)$, that is, $J(P_c)$, we have $G_c(z)\to0$. Extend $G_c$ to the whole plane by defining it to be zero elsewhere.

% Source page 13
One can show that $G_c(z)$ is jointly continuous in $c$ and $z$, via some basic calculus. Notice that
\[
P_c(0)=c\quad\Longrightarrow\quad G_c(c)=2G_c(0)>G_c(0).
\]
Thus $\varphi_c(c)$ is well-defined and analytic in $c$. As $c\to\M$, say $c_n\to c_0\in\M$, we have $G_{c_n}(z)\to G_{c_0}(z)$ uniformly in $\C$, so
\[
G_{c_n}(c_n)\longrightarrow G_{c_0}(c_0)=0
\qquad\bigl(\text{as }c_0\in\partial\widetilde A_{c_0}(\infty)\bigr).
\]
Consequently, $|\varphi_c(c)|\to1$ as $c\to\M$. We therefore have an analytic map, with a simple pole at $\infty$,
\[
\Chat\setminus\M\longrightarrow\Chat\setminus\D,
\qquad c\longmapsto\varphi_c(c).
\]
If $\varphi_c(c)$ were not surjective, there would be a sequence $\{c_n\}_n$ with $c_n\to\M$ but $|\varphi_{c_n}(c_n)|\not\to1$. Let $z_0\in\C\setminus\D$. We know that
\[
\frac{1}{2\pi i}\int_\gamma
\frac{\varphi_c'(c)}{\varphi_c(c)-z_0}\,dc
\]
is continuous in $z_0$. By the argument principle, it must be $-1$, because $\Chat\setminus\M$ is connected.\hfill$\square$
