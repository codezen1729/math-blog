\section*{Mixing, Products, and Spectral Criteria}
% BLOG-CONTENT-BEGIN
\subsection{Mixing}
Ergodic theorems connect orbit averages with equidistribution. Mixing asks a
different question: how do measurable sets become statistically independent
under iteration? Three complementary perspectives will be useful:
\begin{itemize}
 \item Measure-theoretic: to obtain dynamical information.
 \item Operator-theoretic: to obtain equidistribution theorems.
 \item Fourier-analytic: for some examples.
\end{itemize}
Consider the ``set-orbits''
$\{T^{\circ n}(E)\}_{n\in\mathbb Z}$. The measure-theoretic perspective
gave us information about $\{T^{\circ n}(x)\}_{n\in\mathbb Z}$, namely
recurrence theorems; the operator-theoretic perspective told us about
$\{f\circ T^n\}_{n\in\mathbb Z}$. We now view the system
$(X,\mathcal B,\mu,T)$ probabilistically. The starting observation is that
\[
 \frac1N\sum_{n=0}^{N-1}f\circ T^n\longrightarrow\int f\,d\mu
 \quad\text{in $L^2_\mu$, as $N\to\infty$.}
\]
One can restate this slightly to obtain a probabilistic interpretation
% Source page 2
of ergodicity:
\[
 T\text{ is ergodic}
 \quad\Longleftrightarrow\quad
 \frac1N\sum_{n=0}^{N-1}\mu(A\cap T^{-n}(B))
 \longrightarrow\mu(A)\mu(B).
\]
\begin{itemize}
 \item If $T$ is ergodic, then, by the dominated convergence theorem,
 \[
  \frac1N\sum_{n=0}^{N-1}\langle f\circ T^n,g\rangle
  \longrightarrow\left(\int f\,d\mu\right)\left(\int g\,d\mu\right).
 \]
 Set $f=\chi_B$ and $g=\chi_A$ to obtain the desired result.
 \item Conversely, if $T^{-1}(A)=A$, set $B=X\setminus A$. We obtain
 $\mu(A)\mu(X\setminus A)=0$, so $\mu(A)\in\{0,1\}$.
\end{itemize}
Thus ergodicity gives asymptotic independence only after averaging. Mixing
strengthens this to $\mu(A\cap T^{-n}(B))\to\mu(A)\mu(B)$ without averaging.

\setcounter{definition}{5}\begin{definition}[Mixing]
A system $(X,\mathcal B,\mu,T)$ is \emph{mixing} if, for all
$A,B\in\mathcal B$,
\[
 \mu(A\cap T^{-n}(B))\longrightarrow\mu(A)\mu(B)
 =\mu(A)\mu(T^{-n}(B))\qquad(n\to\infty).
\]
\end{definition}
\setcounter{definition}{6}\begin{definition}[Weak mixing]
A system $(X,\mathcal B,\mu,T)$ is \emph{weakly mixing} if, for all
$A,B\in\mathcal B$,
\[
 \frac1N\sum_{n=0}^{N-1}
 \bigl|\mu(A\cap T^{-n}(B))-\mu(A)\mu(B)\bigr|
 \longrightarrow0\qquad(N\to\infty).
\]
\end{definition}

% Source page 3
\begin{remark}
For a sequence $\{a_n\}_n$ (for example, $a_n$ may be
$\mu(A\cap T^{-n}B)$),
\[
 \lim_{n\to\infty}a_n=0
 \quad\Longrightarrow\quad
 \lim_{n\to\infty}\frac1n\sum_{i=0}^{n-1}|a_i|=0,
\]
but not conversely. Thus mixing implies weak mixing. Further, given a
weakly mixing system $(X,\mathcal B,\mu,T)$ and $T^{-1}B=B$, we have
\[
 \frac1N\sum_{n=0}^{N-1}|\mu(A\cap B)-\mu(A)\mu(B)|\longrightarrow0.
\]
Set $A=X\setminus B$. Then $\mu(A)\mu(B)\to0$, so $\mu(A)=0$ or
$\mu(B)=0$, and $T$ is ergodic. Consequently,
\[
 \text{Mixing}\ \subset\ \text{Weak mixing}\ \subset\ \text{Ergodicity}.
\]
Said differently:
\begin{itemize}
 \item $T^{-1}$ makes $A$ and $B$ asymptotically independent:
 this is mixing.
 \item $T^{-1}$ makes $A$ and $B$ asymptotically independent except on
 a zero-density set: this is weak mixing.
 \item $T^{-1}$ makes $A$ and $B$ independent on average:
 this is ergodicity.
\end{itemize}
\end{remark}
\begin{remark}
Since $T^{-1}$ separates events in the senses described above, we say that
$T$ mixes events. Hence we
call such systems mixing or weakly mixing. Weak mixing has several equivalent formulations, one of which is central to ergodic proofs of Szemer\'edi's theorem.
\end{remark}
We now turn to equivalent formulations of weak mixing and the connections
% Source page 4
between its probabilistic, operator-theoretic, and measure-theoretic forms.

\setcounter{theorem}{14}\begin{theorem}[Mixing equivalences]
\label{thm:l4-mixing-equivalences}
Given a system $(X,\mathcal B,\mu,T)$, the following are equivalent:
\begin{enumerate}[label=(\arabic*)]
 \setcounter{enumi}{0}\item $T$ is weakly mixing.
 \setcounter{enumi}{1}\item $T\times T$ is ergodic with respect to $\mu\times\mu$.
 \setcounter{enumi}{2}\item $T\times T$ is weakly mixing with respect to $\mu\times\mu$.
 \setcounter{enumi}{3}\item If $(Y,\mathcal B',\nu,S)$ is ergodic, then
 $(X\times Y,\mathcal B\times\mathcal B',\mu\times\nu,T\times S)$
 is ergodic.
 \setcounter{enumi}{4}\item $U_T$ has \emph{continuous spectrum}: its only eigenvalue is $1$,
 and the corresponding eigenspace consists of constant functions.
\end{enumerate}
\end{theorem}
The proof uses the following lemma of Koopman and von Neumann.

\setcounter{theorem}{15}\begin{lemma}[Koopman--Neumann]
\label{lem:l4-koopman-neumann}
Let $\{a_n\}$ be a sequence of bounded nonnegative real numbers.
The following are equivalent:
\begin{enumerate}[label=(\roman*)]
 \setcounter{enumi}{0}\item $\displaystyle\lim_{n\to\infty}\frac1n\sum_{j=0}^{n-1}a_j=0$.
 \setcounter{enumi}{1}\item There exists $J\subseteq\mathbb N$ of density zero, where
 \[
  \operatorname{density}(J):=
  \lim_{n\to\infty}\frac1n|J\cap[1,n]|,
 \]
 such that $a_n\to0$ as $n\to\infty$ with $n\notin J$.
 \setcounter{enumi}{2}\item $\displaystyle\lim_{n\to\infty}\frac1n\sum_{j=0}^{n-1}a_j^2=0$.
\end{enumerate}
\end{lemma}

% Source page 5
\begin{proof}
Set $J_k:=\{j\in\mathbb N:a_j>1/k\}$. This gives the natural inclusions
$J_1\subseteq J_2\subseteq J_3\subseteq\cdots$.
By definition,
\[
 \frac1k|J_k\cap[0,n)|
 <\sum_{\substack{0\leq i\leq n-1\\a_i>1/k}}a_i
 \leq\sum_{i=0}^{n-1}a_i.
\]
Dividing by $n$,
\[
 \frac1n|J_k\cap[0,n)|
 <k\left(\frac1n\sum_{i=0}^{n-1}a_i\right)\longrightarrow0,
\]
so $\operatorname{density}(J_k)=0$ for all $k\geq1$.
Choose $0<\ell_1<\ell_2<\cdots$ such that
$\frac1n|J_k\cap[0,n)|\leq1/k$ for all $n\geq\ell_k$, and set
\[
 J=\bigcup_{k=0}^\infty\bigl(J_k\cap[\ell_k,\ell_{k+1})\bigr).
\]
% Author check: the source starts this union at k=0 although J_k and ell_k were introduced for k>=1.
Since $J_k\subseteq J_{k+1}$,
$J_k\cap[\ell_{k+1},\ell_{k+2})\subseteq
 J_{k+1}\cap[\ell_{k+1},\ell_{k+2})$, so
$J_k\cap[\ell_k,\infty)\subseteq J$.
If $n\notin J$ and $n\geq\ell_k$, then $n\notin J_k$, and hence
$a_n\leq1/k$. Thus $a_n\to0$ with $n\notin J$.
Let $n\in[\ell_k,\ell_{k+1})$. Then
\[
 \begin{aligned}
 J\cap[0,n)
 &=\left(\bigcup_{i=0}^\infty J_i\cap[\ell_i,\ell_{i+1})\right)\cap[0,n)\\
 &\subseteq\left(\bigcup_{i=0}^{k}J_i\cap[\ell_i,\ell_{i+1})\right)\cap[0,n)
 \subseteq J_k\cap[0,n),
 \end{aligned}
\]
because $J_i\subseteq J_{i+1}$ for all $i\geq0$.
Therefore $\frac1n|J\cap[0,n)|\leq\frac1n|J_k\cap[0,n)|\leq1/k$,
by our choice of $\{\ell_k\}_{k\geq0}$. Thus $J$ has density zero.

Now suppose $a_n\to0$ with $n\notin J$. Since $\{a_n\}_n$ is bounded,
there exists $R>0$ such that $a_n\leq R$ for every $n\geq1$.
% Source page 6
For each $k\geq1$, choose $N_k$ such that $n\notin J$, $n\geq N_k$
imply $a_n<1/k$, and such that $\frac1n|J\cap[0,n)|\leq1/k$.
For all $n\geq kN_k$,
\[
 \begin{aligned}
 \frac1n\sum_{i=0}^{n-1}a_i
 &=\frac1n\left(\sum_{i=0}^{N_k-1}a_i
   +\sum_{\substack{i\in J\\N_k\leq i<n}}a_i
   +\sum_{\substack{i\notin J\\N_k\leq i<n}}a_i\right)\\
 &<\frac1n\left(RN_k+R|J\cap[0,n)|+\frac nk\right)\\
 &\leq\frac{RN_k}{n}+R\frac{|J\cap[0,n)|}{n}+\frac1k
 \leq\frac Rk+\frac Rk+\frac1k
 =\frac{2R+1}{k}\longrightarrow0\qquad(k\to\infty).
 \end{aligned}
\]
Similarly,
\[
 a_n\xrightarrow[n\notin J]{}0
 \quad\Longleftrightarrow\quad a_n^2\xrightarrow[n\notin J]{}0
 \quad\Longleftrightarrow\quad\frac1n\sum_{i=0}^{n-1}a_i^2\longrightarrow0.
\]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:l4-mixing-equivalences}]
We shall use the Koopman--Neumann lemma repeatedly in the equivalent form
\[
 \begin{aligned}
 &\frac1N\sum_{n=0}^{N-1}
  |\mu(A\cap T^{-n}B)-\mu(A)\mu(B)|\longrightarrow0\\
 &\quad\Longleftrightarrow\quad
 \frac1N\sum_{n=0}^{N-1}
  |\mu(A\cap T^{-n}B)-\mu(A)\mu(B)|^2\longrightarrow0\\
 &\quad\Longleftrightarrow\quad
 \mu(A\cap T^{-n}B)\xrightarrow[n\notin J]{}\mu(A)\mu(B),
 \quad\text{where $J$ has density zero.}
 \end{aligned}
\]
\paragraph{$(1)\Longrightarrow(3)$.}
Given $A_1,B_1,A_2,B_2\in\mathcal B$, there are density-zero sets
% Source page 7
$J_1,J_2$ such that
\begin{equation}
 \mu(A_1\cap T^{-n}B_1)\xrightarrow[n\notin J_1]{}\mu(A_1)\mu(B_1),
 \qquad
 \mu(A_2\cap T^{-n}B_2)\xrightarrow[n\notin J_2]{}\mu(A_2)\mu(B_2).
 \tag{$*$}\label{eq:l4-pair-limits}
\end{equation}
% Obvious notation typo: the second B on the handwritten left side lacks its subscript 2.
Set $J=J_1\cup J_2$. Clearly,
\[
 \frac1n|J\cap[0,n)|
 \leq\frac1n|J_1\cap[0,n)|+\frac1n|J_2\cap[0,n)|\longrightarrow0,
\]
so $J$ has density zero. Therefore
\[
 \begin{aligned}
 &\lim_{\substack{n\to\infty\\n\notin J}}
 \bigl|(\mu\times\mu)((A_1\times A_2)\cap(T\times T)^{-n}(B_1\times B_2))\\
 &\hspace{38mm}-(\mu\times\mu)(A_1\times A_2)(\mu\times\mu)(B_1\times B_2)\bigr|\\
 &=\lim_{\substack{n\to\infty\\n\notin J}}
 \bigl|\mu(A_1\cap T^{-n}B_1)\mu(A_2\cap T^{-n}B_2)
 -\mu(A_1)\mu(A_2)\mu(B_1)\mu(B_2)\bigr|=0,
 \end{aligned}
\]
by the definition and \eqref{eq:l4-pair-limits}.


\paragraph{$(3)\Longrightarrow(1)$.}
% Source page 9
If $T\times T$ is weakly mixing, then
\[
 \begin{aligned}
 \lim_{n\to\infty}\frac1n\sum_{i=0}^{n-1}
 \bigl|&(\mu\times\mu)((A\times X)\cap(T\times T)^{-n}(B\times X))\\
 &-(\mu\times\mu)(A\times X)(\mu\times\mu)(B\times X)\bigr|=0.
 \end{aligned}
\]
Therefore
\[
 \lim_{n\to\infty}\frac1n\sum_{i=0}^{n-1}
 |\mu(A\cap T^{-n}B)-\mu(A)\mu(B)|=0,
\]
and $T$ is weakly mixing.
% Author check: both sums above use i as the summation variable but n as the iterate in the source.

\paragraph{$(1)\Longrightarrow(4)$.}
Let $(Y,\mathcal B',\nu,S)$ be an ergodic system, and let $T$ be weakly
mixing. For $A_1,B_1\in\mathcal B$ and $A_2,B_2\in\mathcal B'$,
\[
 \begin{aligned}
 &\frac1N\sum_{n=0}^{N-1}(\mu\times\nu)
 ((A_1\times A_2)\cap(T\times S)^{-n}(B_1\times B_2))\\
 &=\frac1N\sum_{n=0}^{N-1}\mu(A_1\cap T^{-n}B_1)\nu(A_2\cap S^{-n}B_2)\\
 &=\frac1N\sum_{n=0}^{N-1}\mu(A_1)\mu(B_1)\nu(A_2\cap S^{-n}B_2)\\
 &\quad+\frac1N\sum_{n=0}^{N-1}
 [\mu(A_1\cap T^{-n}B_1)-\mu(A_1)\mu(B_1)]\nu(A_2\cap S^{-n}B_2).
 \end{aligned}
\]
Since $S$ is ergodic,
$\frac1N\sum_{n=0}^{N-1}\nu(A_2\cap S^{-n}B_2)\to\nu(A_2)\nu(B_2)$.
Since $T$ is weakly mixing,
$\frac1N\sum_{n=0}^{N-1}|\mu(A_1\cap T^{-n}B_1)-\mu(A_1)\mu(B_1)|\to0$.
% Source page 10
Thus
\[
 \frac1N\sum_{n=0}^{N-1}(\mu\times\nu)
 ((A_1\times A_2)\cap(T\times S)^{-n}(B_1\times B_2))
 \longrightarrow\mu(A_1)\mu(B_1)\nu(A_2)\nu(B_2),
\]
and $T\times S$ is ergodic. A monotone-class argument extends this result from
the algebra of rectangles to the $\sigma$-algebra.

\paragraph{$(4)\Longrightarrow(2)$.}
Let $(Y,\mathcal B',\nu,S)$ be the ergodic system with $Y=\{y\}$,
$T=\id$, and $\nu(\{y\})=1$.
% Author check: the source calls the singleton map T rather than S here.
Then
\[
 (X\times Y,\mathcal B\times\mathcal B',\mu\times\nu,T\times S)
 \cong(X,\mathcal B,\mu,T),
\]
so $T$ is ergodic. Alternatively, as stated in the remark, weak mixing
implies ergodicity. Applying (4) again, $T\times T$ is ergodic.

\paragraph{$(2)\Longrightarrow(1)$.}
Let $\mu^2=\mu\times\mu$ be the product measure. Since $T\times T$ is
ergodic, we have
\[
 \begin{aligned}
 \frac1N\sum_{n=0}^{N-1}\mu(A\cap T^{-n}B)
 &=\frac1N\sum_{n=0}^{N-1}\mu^2((A\times X)\cap(T\times T)^{-n}(B\times X))\\
 &\longrightarrow\mu^2(A\times X)\mu^2(B\times X)=\mu(A)\mu(B),
 \end{aligned}
\]
and
\[
 \begin{aligned}
 \frac1N\sum_{n=0}^{N-1}\mu(A\cap T^{-n}B)^2
 &=\frac1N\sum_{n=0}^{N-1}\mu^2((A\times A)\cap(T\times T)^{-n}(B\times B))\\
 &\longrightarrow\mu^2(A\times A)\mu^2(B\times B)=\mu(A)^2\mu(B)^2.
 \end{aligned}
\]
Now we have
% Source page 11
\[
 \begin{aligned}
 &\frac1N\sum_{n=0}^{N-1}(\mu(A\cap T^{-n}B)-\mu(A)\mu(B))^2\\
 &=\frac1N\sum_{n=0}^{N-1}\mu(A\cap T^{-n}B)^2+\mu(A)^2\mu(B)^2
 -2\mu(A)\mu(B)\left(\frac1N\sum_{n=0}^{N-1}\mu(A\cap T^{-n}B)\right)\\
 &\longrightarrow2\mu(A)^2\mu(B)^2-2\mu(A)\mu(B)\mu(A)\mu(B)=0.
 \end{aligned}
\]

\paragraph{$(2)\Longrightarrow(5)$.}
Suppose $U_Tf=\lambda f$, where $\lambda\in S^1$ because $U_T$ is
unitary. Define $g(x_1,x_2)=f(x_1)\overline{f(x_2)}$. Then
\[
 U_{T\times T}g(x,y)=g(Tx,Ty)=\lambda\overline\lambda\,g(x,y)=g(x,y).
\]
By ergodicity of $T\times T$, $g$ must be constant almost everywhere.
Also, $U_Tf=\lambda f$ gives $f\circ T=\lambda f$, so $\lambda=1$.
Hence $U_T$ has continuous spectrum.


\end{proof}

We finish with the Rokhlin problem.
\setcounter{definition}{7}\begin{definition}[$k$-fold mixing]
A system $(X,\mathcal B,\mu,T)$ is \emph{$k$-fold mixing} if
\[
 \mu(A_0\cap T^{-n_1}A_1\cap\cdots\cap T^{-n_k}A_k)
 \longrightarrow\mu(A_0)\cdots\mu(A_k)
\]
as $n_1,n_2-n_1,n_3-n_2,\ldots,n_k-n_{k-1}\to\infty$, for all
$A_i\in\mathcal B$, $0\leq i\leq k$.
\end{definition}
\begin{question}[Rokhlin problem (open)]
If $(X,\mathcal B,\mu,T)$ is mixing, is it $k$-fold mixing for every
$k\geq1$?
\end{question}
