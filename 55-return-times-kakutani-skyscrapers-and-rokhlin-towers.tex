\section*{Return Times, Kakutani Skyscrapers, and Rokhlin Towers}
% BLOG-CONTENT-BEGIN
The Rokhlin problem is a deep question about mixing systems. The constructions
of Rokhlin towers and Kakutani skyscrapers give a geometric way to organize
return times.

\subsection{Induced Transformations}
Let $(X,\mathcal B,\mu,T)$ be a system in which $T$ is invertible. Poincar\'e's
recurrence theorem says that the orbit of almost every point of a measurable
set $A$ returns to $A$ infinitely often. This leads to a quantitative question:
\begin{quote}
What is the expected return time to $A$? Assume $\mu(A)>0$.
\end{quote}
We address this question through the first return time to $A$. Each later
return is the first return of a suitable iterate of the starting point.

Define
\[
 r_A(x):=\inf_{n\geq1}\{n\mid T^n(x)\in A\}.
\]

% Source page 2
$r_A$ naturally defines a transformation on $A$.
\setcounter{definition}{8}\begin{definition}[Induced transformation]
Define $T_A\colon A\to A$ by $T_A(x):=T^{r_A(x)}(x)$. This is defined almost everywhere on $A$, and is called the transformation \emph{induced by $T$ on $A$}, or the \emph{induced transformation on $A$}.
\end{definition}

\begin{itemize}
\item $r_A\colon X\to\mathbb N$ is measurable, because we have
\[
\begin{aligned}
 \text{``first return'':}\quad \{x\in A\mid r_A(x)=1\}&=A_1=A\cap T^{-1}(A),\\
 \text{``second return'':}\quad \{x\in A\mid r_A(x)=2\}&=A_2=(A\cap T^{-2}(A))\setminus A_1,\\
 &\ \vdots\\
 \text{``$n$th return'':}\quad \{x\in A\mid r_A(x)=n\}&=A_n=(A\cap T^{-n}(A))\setminus\bigcup_{i<n}A_i.
\end{aligned}
\]
As $A_n$ is measurable for every $n\geq1$, $r_A$ is measurable, for the sets $\{n\}$ generate the $\sigma$-algebra on $\mathbb N$.
% Source writes X -> N although r_A may be infinite outside the returning set; retained.

\item $T_A\colon A\to A$ is also measurable. Since $T$ is invertible, for $B\in\mathcal B$ with $B\subseteq A$, clearly $T_A^{-1}(B)=\widetilde B$ almost everywhere, so $T_A$ is measurable. Here notice that
\[
 \widetilde B=\bigcup_{n\geq1}T^{-n}(B\cap A_n)
 \quad\text{almost everywhere},
\]
as $T^{-1}$ is defined almost everywhere.
% Author check: the source places B cap A_n inside T^{-n}; retained exactly.
\end{itemize}

% Source page 3
\begin{itemize}
\item $\bigl(A,\mathcal B|_A,\mu_A=\frac1{\mu(A)}(\mu|_A),T_A\bigr)$ is a measure-preserving system.
\item As one would expect, if $T$ is ergodic, then so is $T_A$. First, notice that $A=\bigsqcup_{n\geq1}A_n$. Thus, if $B\in\mathcal B|_A$, then $B=\bigsqcup_{n\geq1}(B\cap A_n)$, so
\[
 \mu_A(B)=\frac1{\mu(A)}\sum_{n\geq1}\mu(B\cap A_n),
 \qquad
 T_A(B)=\bigsqcup_{n\geq1}T_A(B\cap A_n)
       =\bigsqcup_{n\geq1}T^n(B\cap A_n).
\]
Hence
\[
\begin{aligned}
 \mu_A(T_A(B))
 &=\frac1{\mu(A)}\sum_{n\geq1}\mu\bigl(T^n(B\cap A_n)\bigr)\\
 &=\frac1{\mu(A)}\sum_{n\geq1}\mu(B\cap A_n)
   &&\text{($T$ is $\mu$-invariant)}\\
 &=\mu(B).
\end{aligned}
\]
% Author check: source concludes mu(B), not mu_A(B); no mathematical correction made.
If $T_A$ is not ergodic, there exists $B$ with $T_A^{-1}(B)=B$ and $0<\mu(B)<\mu(A)$, i.e. $0<\mu_A(B)<1$. Then
\[
 \bigcup_{n\geq1}\ \bigcup_{j=0}^{n-1}T^j(B\cap A_n)
\]
is $T$-invariant and has nonzero measure.
\end{itemize}

The induced transformation leads naturally to Kakutani skyscrapers. This construction yields two useful results, including the expected-return-time formula.

\subsection{Kakutani Skyscrapers}
\begin{figure}[htbp]
 \centering
 \includegraphics[width=.92\textwidth]{lecture5-fig-01.pdf}
 \FigureTag{L5.1}{fig:lecture5-fig-01}
\end{figure}

\paragraph{Description.}
The ground floor represents $A$ as the disjoint union of the sets
$\{A_k\}_{k\geq1}$. A point of $A_k$ returns to $A$ after $k$ iterates, so
$A_k,T(A_k),\ldots,T^{k-1}(A_k)$ form a tower of height $k$. The map $T$
moves a point up one floor at a time, while $T_A$ records its eventual return
to the base.

\begin{remark}
Poincar\'e recurrence implies that the ground floor has $\{A_k\}_{k\geq1}$ almost everywhere, and hence the tower above it. Ergodicity of $T$ implies that every point of $X$ belongs to some floor of the skyscraper, as $\mu(A)>0$.
\end{remark}

% Source page 5
\setcounter{theorem}{16}\begin{theorem}[Expected Return Time, due to Kac]\label{thm:l5-kac}
Let $(X,\mathcal B,\mu,T)$ be an invertible ergodic system. If $\mu(A)>0$, then
\[
 \int_A r_A\,d\mu=1.
\]
In other words, the expected return time is
\[
 \frac1{\mu(A)}\int_A r_A\,d\mu=\frac1{\mu(A)}.
\]
\end{theorem}
\Needspace{5\baselineskip}
\begin{proof}
Since $T$ is invertible, ergodicity of $T$ implies ergodicity of $T^{-1}$.  $\mu\bigl(\bigcup_{n\geq1}T^n(A)\bigr)=1$, so the skyscraper contains almost all of $X$. Thus, summing up tower-wise,
\[
\begin{aligned}
 1=\mu(X)
 &=\sum_{n\geq1}\mu\bigl(A_n\sqcup T(A_n)\sqcup\cdots\sqcup T^{n-1}(A_n)\bigr)\\
 &=\sum_{n\geq1}n\mu(A_n)=\int_A r_A\,d\mu,
\end{aligned}
\]
as $\sum_{k=1}^n k\chi_{A_k}\nearrow r_A$.
\end{proof}
Thus larger sets have shorter expected return times. In particular, if $\mu(A)>\frac12$, then the expected return time is less than $2$.

\setcounter{theorem}{17}\begin{theorem}[Rokhlin's Tower Decomposition]\label{thm:l5-rokhlin}
In the setting of Theorem~\ref{thm:l5-kac}, suppose that $\mu(\{x\})=0$ for all $x\in X$, i.e. $\mu$ is nonatomic. Then for every $n\geq1$ and $\varepsilon>0$, there exists $B\in\mathcal B$ such that
\begin{itemize}
 \item $B,T(B),\ldots,T^{n-1}(B)$ are disjoint;
 \item $\displaystyle\mu\left(\bigsqcup_{k=0}^{n-1}T^k(B)\right)>1-\varepsilon$.
\end{itemize}
\end{theorem}

% Source page 6
\begin{proof}
A nonatomic measure has no ``atoms''. More precisely, given $E\in\mathcal B$,
there exists $F\in\mathcal B$ such that $0<\mu(F)<\mu(E)$. Then $\mu(F)$ or
$\mu(E\setminus F)$ is less than $\mu(E)/2$. Iterating this construction gives
measurable sets satisfying
\[
 \mu(\widetilde F)<\frac{\mu(E)}{2^n}<\frac1{2^n}.
\]
% Author check: source says ``replacing F by E'' and omits F subset E; retained.
Now, given $n$ and $\varepsilon$, select $A\in\mathcal B$ such that $0<\mu(A)<\varepsilon/n$. The skyscraper yields the decomposition of $X$ into towers
\[
 A_k\sqcup T(A_k)\sqcup\cdots\sqcup T^{k-1}(A_k).
\]
Set
\[
 B=\bigsqcup_{k\geq n}\ \bigsqcup_{j=0}^{\lfloor k/n\rfloor-1}T^{jn}(A_k).
\]
Then $B,T(B),T^2(B),\ldots,T^{n-1}(B)$ are disjoint; each belongs to a particular residue class modulo $n$. The set $\bigcup_{k=0}^{n-1}T^k(B)$ is all of $X$ except those sets in the skyscraper comprising no more than $n$ floors in each tower. Therefore
\[
 \mu\left(\left(\bigsqcup_{k=0}^{n-1}T^k(B)\right)^c\right)
 \leq n\sum_{k=1}^{\infty}\mu(A_k)
 \leq n\mu(A)<\varepsilon,
\]
which implies $\mu\bigl(\bigsqcup_{k=0}^{n-1}T^k(B)\bigr)>1-\varepsilon$.
\end{proof}

Since we are stopping at $\lfloor k/n\rfloor-1$, all towers between $A_4$ and $A_8$ do not reach floors beyond $4$. Similarly, towers between $A_8$ and $A_{12}$ do not reach beyond the eighth floor, and so on. Each tower has residual floors not covered by iterates of $B$, and there can be at most $4$ in our case.

This decomposition is usually called a \emph{Rokhlin tower of height $n$ with base $B$ and residual set of measure less than $\varepsilon$}.

Two fundamental questions now emerge:

% Source page 8
\begin{enumerate}
 \setcounter{enumi}{0}\item The problem of decomposing the state space into ergodic components.
 \setcounter{enumi}{1}\item The problem of classifying measure-preserving systems.
\end{enumerate}
They lead respectively to ergodic decomposition and the classification of
measure-preserving systems, and ultimately towards ergodic approaches to
Szemer\'edi's theorem.
