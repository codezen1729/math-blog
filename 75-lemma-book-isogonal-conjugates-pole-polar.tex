\subsection{Isogonal conjugates}
% Source page 2 - right (left is inside cover)
We begin with the geometry of isogonal conjugates.
\setcounter{theorem}{97}\begin{lemma}[Isogonal conjugates]\label{lemma:lb2-101}
Given a point $P$ inside triangle $ABC$, reflect $AP,BP,CP$ in the angle bisectors of $A,B,C$.
The resulting lines are concurrent. Their point of concurrence is the isogonal conjugate of $P$ in triangle $ABC$.
\end{lemma}
\begin{figure}[H]\centering
\includegraphics[width=.67\linewidth]{lb2-fig-01.pdf}
\FigureTag{LB2.1}{fig:lb2-01}
\end{figure}
The circumcentre and orthocentre are isogonal conjugates of each other, while the incentre is its own isogonal conjugate.
\paragraph{Application.}
The following IMO problem gives a natural first application.
% Source page 3 - left
\paragraph{IMO (1996).}
In triangle $ABC$, an interior point $P$ satisfies
$\angle APB-\angle C=\angle APC-\angle B$.
Prove that the bisectors of $\angle PBA$ and $\angle PCA$, together with $AP$, concur.
\begin{figure}[H]\centering
\includegraphics[width=.84\linewidth]{lb2-fig-02.pdf}
\FigureTag{LB2.2}{fig:lb2-02}
\end{figure}
The angle condition on $P$ gives $\alpha+\delta=\beta+\gamma$.
Angle bisectors concur to give the incentre; therefore $AP$ and the bisectors of $\angle ABP$ and $\angle ACP$ are concurrent.
This proves the required concurrence. The two points shown on $AP$ need not coincide.
\paragraph{Pythagoras Olympiad (1980).}
In triangle $ABC$, a point $D$ satisfies
$\angle DCA=\angle DCB=\angle DBC=10^\circ$ and $\angle DBA=20^\circ$.
Find $\angle CAD$.
\begin{figure}[H]\centering
\includegraphics[width=.95\linewidth]{lb2-fig-03.pdf}
\FigureTag{LB2.3}{fig:lb2-03}
\end{figure}
Put $x=\angle CAD$. Applying trigonometric Ceva gives
\[
\frac{\sin x}{\sin(130^\circ-x)}\frac{\sin20^\circ}{\sin10^\circ}
\frac{\sin10^\circ}{\sin10^\circ}=1,
\]
and hence
\[
\frac{\sin x}{\sin(130^\circ-x)}
=\frac{\sin10^\circ}{\sin20^\circ}
=\frac{\sin30^\circ}{\sin100^\circ}.
\]
The function $t\mapsto\sin t/\sin(130^\circ-t)$ is strictly increasing for
$0<t<130^\circ$, so $x=30^\circ$.

\setcounter{theorem}{98}\begin{lemma}[Concurrent perpendiculars]\label{lemma:lb2-102}
Let $A,B,C$ form a triangle and $P,Q,R$ be three points in the plane.
The lines through $P,Q,R$ perpendicular to $BC,CA,AB$, respectively, are concurrent if and only if
$BP^2-PC^2+CQ^2-QA^2+AR^2-RB^2=0$.
\end{lemma}
% Source page 4 - left
This has a crisp application.
\paragraph{IMOTC: Indian Team Selection Test 3 (2014).}
For $i=1,2,3$, let $x_i,y_i$ be nonzero real numbers, and put $v_i=x_i+y_i$.
Suppose
\[
x_1x_2x_3=-y_1y_2y_3,\qquad x_1^2+x_2^2+x_3^2=y_1^2+y_2^2+y_3^2,
\]
and both $(v_1,v_2,v_3)$ and $(v_1^2,v_2^2,v_3^2)$ satisfy the triangle inequalities.
Prove that exactly one of $x_1,x_2,x_3,y_1,y_2,y_3$ is negative.
Let $ABC$ have $BC=v_1$, $CA=v_2$, $AB=v_3$.
The triangle inequalities for the squares imply that $ABC$ is acute.
Take $D,E,F$ on the extended sides $BC,CA,AB$, respectively, with directed segments
$\overline{BD}=x_1$, $\overline{CE}=x_2$, $\overline{AF}=x_3$.
The product condition and Menelaus's theorem imply that $D,E,F$ are collinear.
The squared-sum condition implies that the perpendiculars at $D,E,F$ to $BC,CA,AB$ concur.
Thus $D,E,F$ form a Simson line, and their perpendiculars concur on the circumcircle.
Since $ABC$ is acute, exactly one lies outside the corresponding side segment, proving that exactly one number is negative.

\setcounter{theorem}{102}\begin{lemma}[Trigonometric Ceva]
In triangle $ABC$, the lines $AD,BE,CF$ are concurrent if and only if
$\sin\angle BAD\sin\angle ACF\sin\angle CBE
=\sin\angle DAC\sin\angle BCF\sin\angle ABE$.
\end{lemma}
The lemma is useful in both concurrency and collinearity arguments.
\paragraph{Application: British MO 2013--14, round 2.}
The point $P$ lies inside $ABC$ with $\angle ABP=\angle PCA$.
Choose $Q$ so that $PBQC$ is a parallelogram. Prove $\angle QAB=\angle CAP$.
\begin{figure}[H]\centering
\includegraphics[width=.70\linewidth]{lb2-fig-08.pdf}
\FigureTag{LB2.8}{fig:lb2-08}
\end{figure}
This solution is due to Pranjal Warade.
It is equivalent to proving that the isogonal conjugate $P'$ of $P$ lies on $AQ$;
in other words, that $A,P',Q$ are collinear, or that $AQ,BB',CC'$ concur.
Trigonometric Ceva gives
\[
\frac{\sin\angle QAB}{\sin\angle QAC}
\frac{\sin\angle ACC'}{\sin\angle BCC'}
\frac{\sin\angle CBB'}{\sin\angle ABB'}
=\frac{\sin\angle QAB}{\sin\angle QAC}
\frac{\sin\angle PCB}{\sin\angle PBC}
=\frac{BQ}{QC}\frac{BP}{PC}=1.
\]

\subsection{Pole and polar argument}
The pole--polar method begins with the standard definitions.
\begin{figure}[H]\centering
\includegraphics[width=.70\linewidth]{lb2-fig-09.pdf}
\FigureTag{LB2.9}{fig:lb2-09}
\end{figure}
In the drawing, $P$ is the pole of $XY$ and the line $p=XY$ is the polar of $P$.
The method rests on three useful properties.
\setcounter{theorem}{103}\begin{lemma}[La Hire's theorem: PAPA I]\label{lemma:lb2-106}
Let $x,y$ be the polars of $X,Y$, respectively. Then $X$ lies on $y$ if and only if $Y$ lies on $x$.
\end{lemma}
\setcounter{theorem}{104}\begin{lemma}[Intersection of polars: PAPA II]\label{lemma:lb2-107}
Let $x,y,z$ be the polars of $X,Y,Z$. Then $Z=x\cap y$ if and only if $z=XY$.
\end{lemma}
\setcounter{theorem}{105}\begin{lemma}[Complete quadrilateral: PAPA III]\label{lemma:lb2-108}
Let $W,X,Y,Z$ lie on a circle. If $P=XY\cap WZ$, $Q=WX\cap ZY$, and $R=XZ\cap YW$,
then the polar of $P$ is $QR$. Similarly, the polar of $Q$ is $PR$.
\end{lemma}
% Source page 12 - left
The following applications use the diagram-first style of Akopyan's \emph{Geometry in Figures}.
\Needspace{29\baselineskip}
\paragraph{Problem 1.}
Prove $RS\perp UV$.
\begin{figure}[H]\centering
\includegraphics[width=.78\linewidth]{lb2-fig-10.pdf}
\FigureTag{LB2.10}{fig:lb2-10}
\end{figure}
$K$ lies on $r$, so $R$ lies on $k$ by PAPA I, La Hire's theorem.
$S$ lies on $k$ by PAPA III, so $k=RS$ and $RS\perp UV$.

This problem appeared in the Chinese Mathematical Olympiad.
% Source page 12 - right
\paragraph{Problem 2.}
Prove that $P,E,F$ are collinear.
\begin{figure}[H]\centering
\includegraphics[width=.78\linewidth]{lb2-fig-11.pdf}
\FigureTag{LB2.11}{fig:lb2-11}
\end{figure}
By definition, $q=EF$. By PAPA III, $P$ lies on $q$.
Therefore $P,E,F$ are collinear.
This problem appeared in the 1997 Chinese Mathematical Olympiad.
