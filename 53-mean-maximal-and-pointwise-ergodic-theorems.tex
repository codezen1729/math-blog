\section*{Mean, Maximal, and Pointwise Ergodic Theorems}
% BLOG-CONTENT-BEGIN
Poincar\'e recurrence tells us that $\{T^{\circ n}(x)\}_n$ is recurrent.
If $T$ is ergodic, then for almost every $x\in X$ this orbit returns to every
positive-measure subset of $X$. The next question is quantitative: how often
does it return? More precisely, we ask
what happens, as $N\to\infty$, to
\[
  A_N^f:=\frac1N\sum_{n=0}^{N-1}U_T^n f
  \qquad\text{(the $T$-average of $f$).}
\]
In answering what happens to $\frac1{N+1}\sum_{n=0}^{N}U_T^n f$ as
$N\to\infty$, ergodic theorems will, in some sense, establish a duality
between space and time. The main result is the following.

\setcounter{theorem}{8}\begin{theorem}[Birkhoff]
Let $(X,\mathcal B,\mu,T)$ be a measure-preserving system, with $T$ ergodic.
If $f\in L^1_\mu$, then
\[
  \lim_{n\to\infty}\frac1n\sum_{j=0}^{n-1}U_T^j f=\int f\,d\mu
  \qquad\text{almost everywhere.}
\]
\end{theorem}

% Source page 2
The left-hand side is an average over time, and the right-hand side, being
an integral, is an average over space. If $f=\chi_B$, then the ergodic
theorem is telling you that the probability of $T^n(x)\in B$ is given by
the size of $B$ (or the measure of $B$). The conclusion is independent of $x\in X$, and the imposition that
$T$ be ergodic is essential; otherwise, there may be $x\in X$ such that
$T^n(x)\notin B$ for all $n\geq1$. Removing that invariant-set obstruction allows the measure of a set to govern the recurrence frequency. This is the time--space duality.

For a positive-measure set $B$, ergodicity gives
\[
 \bigcup_{n\geq0}T^{-n}(B)=X\qquad\text{almost everywhere}.
\]

Thus certain spatial information can be translated into temporal information,
and vice versa.

% Source page 3
We now give a complete proof of Birkhoff's ergodic theorem, often also called
the pointwise ergodic theorem. The proof emphasizes its operator-theoretic
nature.

\setcounter{theorem}{9}\begin{theorem}[von Neumann's mean ergodic theorem: $L^2_\mu$ version]
\label{thm:l3-mean-l2}
Let $(X,\mathcal B,\mu,T)$ be a measure-preserving system. Let $P_T$ be the
orthogonal projection of $L^2_\mu$ onto
\[
  I=\{g\in L^2_\mu:U_Tg=g\}.
\]
Then, for every $f\in L^2_\mu$,
\[
  \frac1N\sum_{n=0}^{N-1}U_T^nf\longrightarrow P_Tf
  \quad\text{in the $L^2_\mu$ norm.}
\]
\end{theorem}
\begin{proof}
First determine $I^\perp$. Set
$B=\{U_Tg-g:g\in L^2_\mu\}$, and claim that $B^\perp=I$.
If $f\in I$, then $U_Tf=f$. Hence
\[
 \langle f,U_Tg-g\rangle
 =\langle U_Tf,U_Tg\rangle-\langle f,g\rangle=0,
\]
so $f\in B^\perp$. If $f\in B^\perp$, then
\[
 \langle U_Tg-g,f\rangle=0
 \ \Longrightarrow\ \langle U_Tg,f\rangle=\langle g,f\rangle
 \ \Longrightarrow\ \langle g,(U_T^*-\id)f\rangle=0
 \qquad(g\in L^2_\mu).
\]
Therefore
\[
 \sup_{\|g\|\leq1}|\langle g,(U_T^*-\id)f\rangle|
 =0=\|(U_T^*-\id)f\|.
\]
% Source page 4
Thus $U_T^*f=f$. Since $U_T$ is an isometry,
$\|U_Tf-f\|_2^2=\|U_Tf\|_2^2+\|f\|_2^2-2\operatorname{Re}\langle U_Tf,f\rangle=0$.
Hence $U_Tf=f$, so $f\in I$.
Consequently, $B^\perp=I$ and $L^2_\mu(X)=I\oplus\overline B$.
Write $f=P_Tf+h$, with $h\in\overline B$. It now suffices to prove
$\frac1N\sum_{n=0}^{N-1}U_T^nh\to0$ in $L^2_\mu$.
For $h=U_Tg-g$, with $g\in L^2_\mu$, the telescoping series gives
\[
 \left\|\frac1N\sum_{n=0}^{N-1}U_T^n(U_Tg-g)\right\|_2
 =\frac1N\|U_T^Ng-g\|_2
 \leq\frac{2\|g\|_2}{N}.
\]
Here $U_T$, being an isometry, must have norm $1$. Therefore
$\frac1N\sum_{n=0}^{N-1}U_T^nh\to0$ in $L^2_\mu$ for $h\in B$, since
$2\|g\|_2/N\to0$ as $N\to\infty$. If $h\in\overline B$, choose $b\in B$
with $\|h-b\|_2<\varepsilon$. Since every $U_T^n$ is an isometry,
\[
 \left\|\frac1N\sum_{n=0}^{N-1}U_T^nh\right\|_2
 \leq\varepsilon+
 \left\|\frac1N\sum_{n=0}^{N-1}U_T^nb\right\|_2.
\]
Letting first $N\to\infty$ and then $\varepsilon\to0$ completes the proof.
\end{proof}

\setcounter{theorem}{10}\begin{theorem}[von Neumann's mean ergodic theorem: $L^1_\mu$ version]
\label{thm:l3-mean-l1}
Let $(X,\mathcal B,\mu,T)$ be a measure-preserving system. For every
$f\in L^1_\mu$,
\[
 \frac1N\sum_{n=0}^{N-1}U_T^nf\longrightarrow f'\in L^1_\mu
 \quad\text{in $L^1_\mu$,}\qquad U_Tf'=f'.
\]
\end{theorem}
\begin{proof}
Note that
\[
 \left(\int_X|f|\,d\mu\right)^2
 \leq\left(\int_X|f|^2\,d\mu\right)
       \left(\int_X1^2\,d\mu\right),
 \qquad\text{so }\|\cdot\|_1\leq\|\cdot\|_2.
\]
If $g\in L^\infty_\mu\subset L^2_\mu$, then
$\frac1N\sum_{n=0}^{N-1}U_T^ng\to g'$ in $L^1_\mu$.
Now $\|A_N^g\|_\infty\leq\|g\|_\infty$, and
$|\langle A_N^g,\chi_B\rangle|\leq\|g\|_\infty\mu(B)$.
Since $A_N^g\to g'$, the dominated convergence theorem gives
$|\langle g',\chi_B\rangle|\leq\|g\|_\infty\mu(B)$.

% Source page 5
If $g'$ does not have bounded essential supremum, one can find sets $B_N$
such that $\mu(B_N)>0$ and $g'|_{B_N}>N$. Then
\[
 N\mu(B_N)<|\langle g',\chi_{B_N}\rangle|
 \leq\|g\|_\infty\mu(B_N),
\]
a contradiction. Thus $g'\in L^\infty_\mu$. As $g\in L^2_\mu$, the
$L^2$ version of von Neumann's ergodic theorem tells us that $g'$ is
$T$-invariant.

Now we exploit the fact that $L^\infty_\mu$ is dense in $L^1_\mu$.
Fix $f\in L^1_\mu$ and $\epsilon>0$. Let $g\in L^\infty_\mu$ satisfy
$\|f-g\|_1<\epsilon$. Since $T$ is $\mu$-invariant,
$\|A_N^f-A_N^g\|_1<\epsilon$. Also, there are $g'$ and $N_0$ such that
$\|A_N^g-g'\|_1<\epsilon$ for all $N\geq N_0$.
By the triangle inequality,
\[
 \|A_N^f-A_{N'}^f\|_1<4\epsilon\qquad(N,N'\geq N_0).
\]
Thus $A_N^f$ is Cauchy in $L^1_\mu$. Since $L^1_\mu$ is complete
(the Riesz--Fischer theorem), $A_N^f\to f'$.

The following identity shows that $f'$ is $T$-invariant:
\[
 U_TA_N^f=\frac{N+1}{N}A_{N+1}^f-\frac1Nf,
\]
and hence
\[
 \|U_TA_N^f-A_N^f\|_1
 =\left\|\frac{N+1}{N}A_{N+1}^f-A_N^f-\frac1Nf\right\|_1
 \leq\|A_{N+1}^f-A_N^f\|_1+\frac2N\|f\|_1.
\]
As $N\to\infty$, $\|f'\circ T-f'\|_1=0$, so $f'\circ T=f'$ almost
everywhere.
\end{proof}

% Source page 6
\begin{remark}
\label{rem:l3-frequency}
If $T$ is ergodic, then, for $f=\chi_B$, we only know so far that the
frequency with which an orbit visits $B$ is a fixed number, independent of $x$.
\end{remark}
\begin{remark}[Quantitative estimate]
We have $A_N^{\chi_B}=\frac1N\sum_{n=0}^{N-1}\chi_B\circ T^n$.
As $T$ is $\mu$-invariant,
\[
 \int_X\chi_B\circ T\,d\mu=\int_X\chi_B\,d\mu=\mu(B)
 \qquad\text{(by Litmus Test 2).}
\]
Therefore
\[
 \int_X A_N^{\chi_B}\,d\mu=\mu(B)
 \geq\sqrt{\mu(B)}\,
 \mu\bigl(\{x:A_N^{\chi_B}(x)>\sqrt{\mu(B)}\}\bigr),
\]
and hence
\[
 \mu\bigl(\{x:A_N^{\chi_B}(x)>\sqrt{\mu(B)}\}\bigr)
 \leq\sqrt{\mu(B)}.
\]
Thus the measure of points in $X$ that spend at least a proportion
$\sqrt{\mu(B)}$ of the time in $B$ is at most $\mu(B)^{1/2}$.
\end{remark}
In light of the quantitative remark above, we have a more quantitative
version of the ergodic theorem.

\setcounter{theorem}{11}\begin{theorem}[Maximal ergodic theorem]
\label{thm:l3-maximal}
Let $(X,\mathcal B,\mu,T)$ be a measure-preserving system, and let
$g:X\to\mathbb R$, $g\in L^1_\mu$. Define, for $\alpha\in\mathbb R$,
\[
 E_\alpha:=\left\{x\in X:
 \sup_{n\geq1}\frac1n\sum_{i=0}^{n-1}g(T^ix)>\alpha\right\}.
\]
Then
\[
 \alpha\mu(E_\alpha)\leq\int_{E_\alpha}g\,d\mu\leq\|g\|_1,
 \qquad
 \alpha\mu(E_\alpha\cap A)\leq\int_{E_\alpha\cap A}g\,d\mu
 \quad\text{if }T^{-1}(A)=A.
\]
\end{theorem}

% Source page 7
The following technical lemma yields the theorem.
\setcounter{theorem}{12}\begin{lemma}
\label{lem:l3-positive}
Let $U:L^1_\mu\to L^1_\mu$ be a positive linear operator with
$\|U\|\leq1$. Given $f\in L^1_\mu$, set
\[
 f_k=\sum_{\ell=0}^{k-1}U^\ell f,\qquad f_0=0,
 \qquad F_N=\max\{f_n:0\leq n\leq N\}.
\]
Then $\int_{\{x:F_N(x)>0\}}f\,d\mu\geq0$ for every $N\geq1$.
\end{lemma}
\begin{proof}
We have $F_N\in L^1_\mu$, since $F_N\leq\sum_{n=0}^N|f_n|$.
Also, $F_N\geq f_n$ for $0\leq n\leq N$, so positivity and linearity give
$UF_N+f\geq Uf_n+f=f_{n+1}$. Thus
$UF_N+f\geq\max_{1\leq n\leq N}f_n$.
Set $P=\{x:F_N(x)>0\}$. For $x\in P$,
\[
 F_N(x)=\max_{0\leq n\leq N}f_n(x)
       =\max_{1\leq n\leq N}f_n(x),
\]
since $f_0=0$. Hence $UF_N(x)+f(x)\geq F_N(x)$, or
$f(x)\geq F_N(x)-UF_N(x)$. By positivity, $UF_N(x)\geq0$, as
$F_N(x)\geq0$. It follows that
\[
 \int_P f\,d\mu
 \geq\int_P F_N\,d\mu-\int_P UF_N\,d\mu
 =\int_XF_N\,d\mu-\int_PUF_N\,d\mu,
\]
% Source page 8
because $F_N(x)=0$ if $x\notin P$. Therefore
\[
 \int_Pf\,d\mu\geq\int_XF_N\,d\mu-\int_XUF_N\,d\mu
 =\|F_N\|_1-\|UF_N\|_1\geq0.
\]
Here the equality uses positivity, and the last inequality uses $\|U\|\leq1$.
\end{proof}

\begin{proof}[Proof of the maximal ergodic theorem]
Apply Lemma~\ref{lem:l3-positive} to $U=U_T$ and $f=g-\alpha$.
If
\[
 P_N:=\left\{x:\max_{1\leq n\leq N}
 \sum_{i=0}^{n-1}(g(T^ix)-\alpha)>0\right\},
\]
then
\[
 0\leq\int_{P_N}(g-\alpha)\,d\mu,
 \qquad \alpha\mu(P_N)\leq\int_{P_N}g\,d\mu.
\]
The sets $P_N$ increase to $E_\alpha$. Letting $N\to\infty$ gives
$\alpha\mu(E_\alpha)\leq\int_{E_\alpha}g\,d\mu\leq\|g\|_1$.
If $T^{-1}(A)=A$, apply the same argument to
$(g-\alpha)\chi_A$. Its positive maximal set is $E_\alpha\cap A$, which
gives the second inequality.
\end{proof}


\begin{remark}
This theorem has essentially removed the dependence on $N$ in the
statement made in the quantitative remark above.
\end{remark}
We now prove Birkhoff's ergodic theorem.

\begin{proof}[Proof of Birkhoff's ergodic theorem]
First consider a function of the form
$h=u+(U_Tg-g)$, where $u\in I$ and $g\in L^2_\mu$. Its averages satisfy
\[
 A_N^h=u+\frac{U_T^Ng-g}{N}.
\]
For every $\varepsilon>0$, invariance of $\mu$ gives
\[
 \sum_{N=1}^{\infty}
 \mu\bigl(\{|U_T^Ng|>\varepsilon N\}\bigr)
 =\sum_{N=1}^{\infty}\mu\bigl(\{|g|>\varepsilon N\}\bigr)<\infty.
\]
The Borel--Cantelli lemma therefore implies $U_T^Ng/N\to0$ almost
everywhere, and so $A_N^h\to u$ almost everywhere. By the $L^2$ mean
ergodic theorem, functions of this form are dense in $L^2_\mu$, hence in
$L^1_\mu$.

For $q\in L^1_\mu$ and $\alpha>0$, the maximal theorem applied to $q$ and
$-q$ gives
\[
 \mu\left(\left\{x:\sup_{N\geq1}|A_N^q(x)|>\alpha\right\}\right)
 \leq\frac{2\|q\|_1}{\alpha}.
\]
Given $f\in L^1_\mu$, approximate it in $L^1$ by the functions $h$ above.
Since $A_N^h$ converges almost everywhere, the last inequality shows that
the upper and lower limits of $A_N^f$ agree almost everywhere. Thus
$A_N^f\to f^*$ almost everywhere.

The $L^1$ mean ergodic theorem also gives $A_N^f\to f'$ in $L^1$, where
$f'$ is $T$-invariant. A subsequence converges to $f'$ almost everywhere,
so $f^*=f'$ almost everywhere. Consequently $f^*$ is $T$-invariant and
\[
 \int_Xf^*\,d\mu=\lim_{N\to\infty}\int_XA_N^f\,d\mu
 =\int_Xf\,d\mu.
\]
\end{proof}

\begin{remark}[Ergodic case]
If $T$ is ergodic, then $f^*$ must be constant, since it is $T$-invariant
(the eigenfunctions of $U_T$ corresponding to $1$ are only the constants).
Thus
\[
 \int f^*\,d\mu=f^*\mu(X)=f^*=\int f\,d\mu,
\]
and we finally get the link between space and time averages:
\[
 \boxed{\lim_{n\to\infty}\frac1n\sum_{i=0}^{n-1}f(T^i(x))
 =\int_Xf\,d\mu\quad\text{for almost every }x.}
\]
\end{remark}
