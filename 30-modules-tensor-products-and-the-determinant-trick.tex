\section*{Modules, Tensor Products, and the Determinant Trick}
% BLOG-CONTENT-BEGIN
\subsection{Modules}
\setcounter{definition}{3}\begin{definition}
$M$ is an \emph{$A$-module} if $M\in\Ob(\mathsf{AbGps})$ is equipped with a unital ring homomorphism $A\to\End_{\Ab}(M)$. If $M'<M$ is an additive subgroup stable under this action, then $M'$ is an \emph{$A$-submodule} of $M$.
\end{definition}
% Source page 11
\begin{remark}
Unwrapping the second condition, say $f:A\to\End(M)$ is a ring homomorphism, gives
\[
f(a)\in\End_{\Ab}(M),\quad f(a+b)=f(a)+f(b),\quad
f(ab)=f(a)f(b),\quad f(1)=1.
\]
Writing $f(a)(x)$ as $ax$, these become
\[
a(x+y)=ax+ay,\quad(a+b)x=ax+bx,\quad(ab)x=a(bx),\quad1x=x.
\]
\end{remark}
\setcounter{example}{3}\begin{example}
\begin{enumerate}[label=(\roman*)]
\setcounter{enumi}{0}\item Ideals of $A$ are $A$-modules.
\setcounter{enumi}{1}\item $G\in\Ob(\mathsf{AbGps})$ is a $\Z$-module.
\setcounter{enumi}{2}\item A $\kk$-vector space is a $\kk$-module.
\setcounter{enumi}{3}\item If $M'\subseteq M$ are $A$-modules, then so is $M/M'$.
\setcounter{enumi}{4}\item If $\{M_i\}_{i\in I}$ are $A$-submodules of $M$, then so is $\sum_{i\in I}M_i$, defined analogously to the sum of ideals.
\setcounter{enumi}{5}\item Products and coproducts exist in the category of $A$-modules. Thus, if the $M_i$ are $A$-modules, so are $\bigoplus_{i\in I}M_i$ (direct sum) and $\prod_{i\in I}M_i$ (direct product).
\end{enumerate}
\end{example}
\setcounter{definition}{4}\begin{definition}
A map $f:M\to M'$ between $A$-modules is a \emph{module homomorphism} if
% Source page 12
$f\in\Hom_{\Ab}(M,M')$ and $f(ax)=af(x)$. Thus $f$ is equivariant with respect to the action of $A$ on $M$ and $M'$.
\end{definition}
\begin{remark}
Let $\AMod$ denote the category of $A$-modules. There are functors
\[
\Hom_{\AMod}(-,N):\AMod^{\mathrm{op}}\to\AMod,\qquad
\Hom_{\AMod}(N,-):\AMod\to\AMod.
\]
\emph{Fact:} These functors are left exact. Thus an exact sequence $0\to M'\to M\to M''\to0$ induces exact sequences
\[
0\to\Hom_A(M'',N)\to\Hom_A(M,N)\to\Hom_A(M',N)
\]
and
\[
0\to\Hom_A(N,M')\to\Hom_A(N,M)\to\Hom_A(N,M'').
\]
One similarly defines right exactness.
\end{remark}

\subsection{Tensor products}
If $M,N$ are $A$-modules, there is a universal object among all $A$-bilinear maps from $M\times N$ to $P$, denoted $M\otimes N$. Thus $f:M\times N\to P$ factors through an $A$-linear map $\widetilde f$, as shown.
\begin{figure}[H]\centering
\includegraphics[width=.48\textwidth]{ca1-fig-06.pdf}
\end{figure}
\paragraph{Construction.}
Consider the direct sum $\bigoplus_{i\in M\times N}A$. Let $C$ be the submodule generated by
\[
\begin{gathered}
(x+x',y)-(x,y)-(x',y),\qquad (x,y+y')-(x,y)-(x,y'),\\
(ax,y)-a(x,y),\qquad(x,ay)-a(x,y).
\end{gathered}
\]
Then $M\otimes N:=\bigl(\bigoplus_{i\in M\times N}A\bigr)/C$.
% The quotient is handwritten above the tensor-product paragraph; the later repeated definition is blank.
It is easy to check that $-\otimes N:\AMod\to\AMod$ and $N\otimes-:\AMod\to\AMod$ are functors. Define $\Psi(f):(x,y)\mapsto f(x)(y)$.
\begin{claim}
\[\Hom(M\otimes N,P)\cong\Hom(M,\Hom(N,P)).\]
\end{claim}
\begin{proof}
% Source page 14
Given $\widetilde\varphi\in\Hom(M\otimes N,P)$, there is a unique $A$-bilinear map $\varphi:M\times N\to P$. Define
\[
\Phi:\Hom(M\otimes N,P)\longrightarrow\Hom(M,\Hom(N,P)),
\qquad \Phi(\widetilde\varphi)(x):=\varphi(x,-).
\]
Similarly, for $f\in\Hom(M,\Hom(N,P))$, the map $\widetilde{\Psi(f)}$ is the inverse of $\Phi$.
% The source leaves the ``where'' line in this inverse construction blank.
\end{proof}
\begin{claim}
The tensor-product functors above are right exact.
\end{claim}
\begin{proof}
The functor $-\otimes N$ is left adjoint to $\Hom(N,-)$, and therefore preserves cokernels. The same argument applies to $N\otimes-$.

More explicitly, if $E'\to E\to E''\to0$ is exact, then for every $P$ the exact sequence
\[
0\longrightarrow\Hom(E'',\Hom(N,P))
\longrightarrow\Hom(E,\Hom(N,P))
\longrightarrow\Hom(E',\Hom(N,P))
\]
identifies, by the tensor--Hom adjunction, with
\[
0\longrightarrow\Hom(E''\otimes N,P)
\longrightarrow\Hom(E\otimes N,P)
\longrightarrow\Hom(E'\otimes N,P).
\]
Thus $E'\otimes N\to E\otimes N\to E''\otimes N\to0$ is exact.
\end{proof}
\subsection{Finitely generated modules}
\setcounter{definition}{5}\begin{definition}
The module $M$ is \emph{free} if $M=\bigoplus_i A$. If $M=\bigoplus_{i=1}^n A$, then $\dim(M)=n$.
\end{definition}
We also have $A\otimes_A A\cong A$ and
$A/\aa\otimes_A A\cong A/\aa$.
\setcounter{definition}{6}\begin{definition}
$N$ is \emph{flat} if $-\otimes N$ is an exact functor. It is already right exact; equivalently, $N$ is flat if
\[
(0\to M'\to M)\ \Longrightarrow\ (0\to M'\otimes N\to M\otimes N).
\]
\end{definition}
\setcounter{theorem}{3}\begin{theorem}[Cayley--Hamilton]
Let $M$ be a finitely generated $A$-module, $\varphi\in\End_{\AMod}(M)$, and $\varphi(M)\subseteq\aa M$, where $\aa$ is an ideal of $A$. Then
\[
\varphi^n+a_1\varphi^{n-1}+\cdots+a_n=0
\]
% Source page 15
for some $a_i\in\aa$, $n>0$.
\end{theorem}
\begin{proof}
Let $\{x_i\}_{i=1}^n$ generate $M$. Since $\varphi(x_i)\in\aa M$, write $\varphi(x_i)=\sum_j a_{ij}x_j$. Equivalently,
\[
\sum_{j=1}^n(\delta_{ij}\varphi-a_{ij})(x_j)=0,
\qquad B\begin{pmatrix}x_1\\\vdots\\x_n\end{pmatrix}=0.
\]
Let $\widetilde B$ be the cofactor matrix of $B$. Then
\[
\widetilde B B\begin{pmatrix}x_1\\\vdots\\x_n\end{pmatrix}=0
\ \Longrightarrow\
\det(B)I\begin{pmatrix}x_1\\\vdots\\x_n\end{pmatrix}=0.
\]
Thus $\det(\delta_{ij}\varphi-a_{ij})\in\End(M)$ is the zero endomorphism.
\end{proof}
\setcounter{theorem}{4}\begin{lemma}[Nakayama]
Let $M$ be a finitely generated $A$-module, and let $\aa$ be an ideal of $A$. If $\aa\subseteq\bigcap_{\mm\text{ maximal}}\mm$ (the Jacobson radical) and $M=\aa M$, then $M=0$.
\end{lemma}
\begin{proof}
Cayley--Hamilton for $\varphi=\id$ yields, for $x=1+a_1+\cdots+a_n\equiv1\pmod\aa$, that $xM=0$. Since $x\equiv1\pmod\aa$,
\[
1-x\in\bigcap_{\mm\text{ maximal}}\mm
\ \Longrightarrow\ (1-x)\in\mm\ \text{for every maximal }\mm
\ \Longrightarrow\ x\in\mm^c\ \text{for every maximal }\mm.
\]
Hence $x$ is a unit. Therefore $M=x^{-1}(xM)=0$.
\end{proof}
