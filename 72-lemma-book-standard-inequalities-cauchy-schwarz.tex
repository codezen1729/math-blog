\subsection{A motivating inequality}
Let $a,b,c>0$ satisfy
$0<\sqrt a+\sqrt b+\sqrt c\leq3\sqrt2$, and establish
\[
\frac{a^2+2a-2}{a\sqrt b}+\frac{b^2+2b-2}{b\sqrt c}
+\frac{c^2+2c-2}{c\sqrt a}\leq3\sqrt2+\ln8.
\]
As written, the challenge is false: $a=1/4$, $b=1$, and $c=729/100$
satisfy the hypothesis, while the left-hand side is approximately $12.652$.

\setcounter{theorem}{70}\begin{lemma}[Arithmetic--geometric mean inequality]\label{lemma:lb1-74}
For all positive reals $a_1,a_2,\ldots,a_n$,
\[\frac{a_1+a_2+\cdots+a_n}{n}\geq\sqrt[n]{a_1a_2\cdots a_n},\]
with equality if $a_1=a_2=\cdots=a_n$.
\end{lemma}
$(1,0,0,\ldots,0)\succ(1/n,1/n,\ldots,1/n)$: direct Muirhead!
Alternatively, consider the constraint $\sum_{i=1}^n a_i=S$ and employ Lagrange's method to prove it.
\paragraph{Application: Nesbitt's inequality.}
For positive $a,b,c$, prove
\[\frac a{b+c}+\frac b{c+a}+\frac c{a+b}\geq\frac32.\]
Set
\[
S=\frac a{b+c}+\frac b{c+a}+\frac c{a+b},\quad
M=\frac b{b+c}+\frac c{c+a}+\frac a{a+b},\quad
N=\frac c{b+c}+\frac a{c+a}+\frac b{a+b}.
\]
Since $M+N=3$, AM--GM gives
\[
M+S=\frac{a+b}{b+c}+\frac{b+c}{c+a}+\frac{c+a}{a+b}\geq3,\qquad
N+S=\frac{a+c}{b+c}+\frac{a+b}{c+a}+\frac{b+c}{a+b}\geq3.
\]
Thus $M+N+2S\geq6$, so $2S\geq3$ and $S\geq3/2$. Q.E.D.!
\setcounter{theorem}{71}\begin{lemma}[Weighted AM--GM inequality]\label{lemma:lb1-75}
For positive reals $a_1,\ldots,a_n$ and nonnegative weights $x_1,\ldots,x_n$, not all zero,
\[\frac{a_1x_1+a_2x_2+\cdots+a_nx_n}{x_1+x_2+\cdots+x_n}
\geq\sqrt[x_1+x_2+\cdots+x_n]{a_1^{x_1}a_2^{x_2}\cdots a_n^{x_n}}.\]
\end{lemma}

% Source page 32 - left
\paragraph{Application: RMO (1989).}
Let $a,b,c$ be the side lengths of a triangle. Prove that
\[(a+b-c)^a(b+c-a)^b(c+a-b)^c\leq a^ab^bc^c.\]
Indeed,
\[
\begin{aligned}
\left(\frac{a+b-c}{a}\right)^a
\left(\frac{b+c-a}{b}\right)^b
\left(\frac{c+a-b}{c}\right)^c
&\leq\left(\frac{a\frac{a+b-c}{a}+b\frac{b+c-a}{b}+c\frac{c+a-b}{c}}{a+b+c}\right)^{a+b+c}\\
&=\left(\frac{2(a+b+c)-(a+b+c)}{a+b+c}\right)^{a+b+c}
=\left(\frac{a+b+c}{a+b+c}\right)^{a+b+c}=1.
\end{aligned}
\]
Q.E.D.
\paragraph{An ingenious application of AM--GM: AMTI Junior, finals (2014).}
For $x,y,z>1$, prove that
\[\frac{x^4}{(y-1)^2}+\frac{y^4}{(z-1)^2}+\frac{z^4}{(x-1)^2}\geq48.\]
We have $x^2/(x-1)\geq4\Longleftrightarrow(x-2)^2\geq0$. Hence
\[
\frac13\left(\frac{x^4}{(y-1)^2}+\frac{y^4}{(z-1)^2}+\frac{z^4}{(x-1)^2}\right)
\geq\sqrt[3]{\left(\frac{x^2}{x-1}\right)^2
\left(\frac{y^2}{y-1}\right)^2\left(\frac{z^2}{z-1}\right)^2}
=\sqrt[3]{4^2\cdot4^2\cdot4^2}.
\]
Therefore $\sum_{\mathrm{cyc}}x^4/(y^2-2y+1)\geq3\cdot4^2=48$. Q.E.D.!
This solution was communicated to me by Vedant. ``Thanks to him!''

\subsection{Standard inequalities}
The following inequalities are useful enough to record as standard lemmas.
\setcounter{theorem}{72}\begin{lemma}[Standard inequalities]\label{lemma:lb1-76}
For positive reals $x,y,z$, the following inequalities hold:
\begin{enumerate}
\setcounter{enumi}{0}\item $(x+y)(y+z)(z+x)\geq8xyz$.
\setcounter{enumi}{1}\item $x^2+y^2+z^2\geq xy+yz+zx$.
\setcounter{enumi}{2}\item $xy+yz+zx\geq\sqrt{3xyz(x+y+z)}$. \emph{Important.}
\setcounter{enumi}{3}\item $x^3+y^3+z^3+5xyz\geq(x+y)(y+z)(z+x)$.
\setcounter{enumi}{4}\item $x^2y^2+y^2z^2+z^2x^2\geq xyz(x+y+z)$. \emph{Important.}
\setcounter{enumi}{5}\item $xy^3+yz^3+zx^3\geq xyz(x+y+z)$.
\end{enumerate}
Also important: $x^2+y^2+z^2\geq x+y+z$ when $xyz=1$.
\end{lemma}
\paragraph{Problems.}
\begin{enumerate}
\setcounter{enumi}{0}\item For positive $x,y,z$, prove $x^2/y^2+y^2/z^2+z^2/x^2\geq x/y+y/z+z/x$ (SAT, 2000).
\setcounter{enumi}{1}\item Find all real triples $(x,y,z)$ satisfying
\[3(x^2+y^2+z^2)=1,\qquad x^2y^2+y^2z^2+z^2x^2=xyz(x+y+z)^3\]
(AMTI Inter, final, 2014).
\setcounter{enumi}{2}\item For positive $a,b$, prove $(1+a)^8+(1+b)^8\geq128ab(a+b)^2$ (IMOTC, 2014).
\setcounter{enumi}{3}\item $R/r\geq b/c+c/b$ (favourite problem at CSL!).
\end{enumerate}
Here is a solution to the third problem, from IMOTC 2014:
\[
(1+a)^8+(1+b)^8\mathrel{\stackrel{?}{\geq}}128ab(a+b)^2
\quad\Longleftrightarrow\quad
\left(\frac{1+a}{2}\right)^8+\left(\frac{1+b}{2}\right)^8
\mathrel{\stackrel{?}{\geq}}\frac{ab(a+b)^2}{2}.
\]
Now
\[
\left(\frac{1+a}{2}\right)^8+\left(\frac{1+b}{2}\right)^8
\geq a^4+b^4\mathrel{\stackrel{?}{\geq}}\frac{ab(a+b)^2}{2}
\quad\Longleftrightarrow\quad
\frac{a^4+b^4}{2}\mathrel{\stackrel{?}{\geq}}ab\left(\frac{a+b}{2}\right)^2.
\]
Since $(a^2+b^2)/2\geq((a+b)/2)^2$ and $ab\leq(a^2+b^2)/2$,
\[
\left(\frac{a^2+b^2}{2}\right)^2\geq ab\left(\frac{a+b}{2}\right)^2,
\qquad
\frac{a^4+b^4}{2}\geq\left(\frac{a^2+b^2}{2}\right)^2
\geq ab\left(\frac{a+b}{2}\right)^2.
\]
The last step uses RMS $\geq$ AM.

The power-mean inequality organizes several familiar comparisons between means.
\setcounter{theorem}{73}\begin{lemma}[Power-mean inequality]\label{lemma:lb1-77}
For positive $a_1,\ldots,a_n$ and $r\geq s>0$,
\[\left(\frac{a_1^r+a_2^r+\cdots+a_n^r}{n}\right)^{1/r}
\geq\left(\frac{a_1^s+a_2^s+\cdots+a_n^s}{n}\right)^{1/s}.\]
\end{lemma}
This is the power-mean inequality. We already dealt with one special case, the ``square-root heuristic''; here is another:
\[\sqrt{\frac{a_1^2+a_2^2+\cdots+a_n^2}{n}}\geq\frac{a_1+a_2+\cdots+a_n}{n}.\]
RMS (root mean square) $\geq$ AM (arithmetic mean).
The next lemma leads to another useful application.
\setcounter{theorem}{74}\begin{lemma}[A sum and its reciprocal sum]\label{lemma:lb1-78}
If $a_1,a_2,\ldots,a_n$ are positive reals, then
\[(a_1+a_2+\cdots+a_n)\left(\frac1{a_1}+\frac1{a_2}+\cdots+\frac1{a_n}\right)\geq n^2.\]
\end{lemma}
To prove it, use AM $\geq$ HM.
\paragraph{An ingenious application, posed by P. K. Hung.}
Let $a,b,c,d$ be positive real numbers. Prove that
\[\left(\frac1a+\frac1b+\frac1c+\frac1d\right)^2
\geq\frac1{a^2}+\frac4{a^2+b^2}+\frac9{a^2+b^2+c^2}
+\frac{16}{a^2+b^2+c^2+d^2}.\]
We have
\[
\frac1{a^2}\geq\frac1{a^2},\qquad
\frac2{ab}\geq\frac4{a^2+b^2},\qquad
\frac2{ac}+\frac2{bc}\geq\frac8{ac+bc},\qquad
\frac1{b^2}+\frac1{c^2}\geq\frac4{b^2+c^2},
\]
and
\[\frac2{ad}+\frac2{bd}+\frac2{cd}
\geq\frac{18}{ad+bd+cd}\geq\frac{16}{a^2+b^2+c^2+d^2}.\]
For the second inequality, Cauchy--Schwarz and AM--GM give
\[
d(a+b+c)\leq\sqrt3d\sqrt{a^2+b^2+c^2}
\leq\frac{\sqrt3}{2}(a^2+b^2+c^2+d^2),
\]
and $8\sqrt3<18$.

% Source page 35 - left
Also,
\[\frac8{bc+ac}\geq\frac8{a^2+b^2+c^2},\qquad
\frac4{b^2+c^2}\geq\frac1{b^2+c^2+a^2}.\]
Now add the results, together with $1/d^2\geq0$, to finish the proof.
\setcounter{theorem}{75}\begin{corollary}[To the reciprocal-sum lemma]
\[\frac x{a_1}+\frac x{a_2}+\cdots+\frac x{a_n}
\geq\frac{xn^2}{a_1+a_2+\cdots+a_n}.\]
This is a form of the modern Cauchy inequality.
\end{corollary}
In fact, P. K. Hung thought of a generalization and conjectured that
\[\left(\frac1{a_1}+\frac1{a_2}+\cdots+\frac1{a_n}\right)^2
\geq\frac1{a_1^2}+\frac4{a_1^2+a_2^2}+\cdots+
\frac{n^2}{a_1^2+a_2^2+\cdots+a_n^2}.\]

\begin{lemma}[Reverse Cauchy]
The best way to apply Cauchy's inequality is what I call the strategy of reverse Cauchy.
Every Olympiad student knows old and modern Cauchy; its modified form is
\[
\frac{x_1^2}{a_1}+\cdots+\frac{x_n^2}{a_n}
\geq\frac{(x_1+\cdots+x_n)^2}{a_1+\cdots+a_n}.
\]
\end{lemma}
\paragraph{Application.}
For $a,b,c>0$, prove that
\[
\frac1{a+b}+\frac1{b+c}+\frac1{c+a}+\frac1{2\sqrt[3]{abc}}
\geq\frac{(a+b+c+\sqrt[3]{abc})^2}{(a+b)(b+c)(c+a)}.
\]
Before employing this divine technique of reverse Cauchy, let us jot down some trivial identities that can be used as lemmas.
\begin{lemma}[Four useful identities]\label{lemma:lb1-29}
\begin{align*}
(a+b)(b+c)(c+a)&=a^2(b+c)+b^2(c+a)+c^2(a+b)+2abc,\\
(a+b)(b+c)(c+a)&=ab(a+b)+bc(b+c)+ca(c+a)+2abc,\\
(a+b-c)(b+c-a)(c+a-b)
&=a(b-a)(a-c)+b(c-b)(b-a)\\
&\quad+c(a-c)(c-b)+abc,\\
(a-b)(b-c)(c-a)&=a^2(c-b)+b^2(a-c)+c^2(b-a).
\end{align*}
\end{lemma}
These are the most celebrated lemmas in the IMO.

\subsection{Q.E.D.s using Cauchy--Schwarz}
\paragraph{Iran (1998).}
Prove that if $x,y,z\geq1$ satisfy $1/x+1/y+1/z=2$, then
\[\sqrt{x-1}+\sqrt{y-1}+\sqrt{z-1}\leq\sqrt{x+y+z}.\]
The usual square-root heuristic does not help here.
In fact, if we make the following adjustment, there is some hope:
\[
\begin{aligned}
\sqrt{x-1}+\sqrt{y-1}+\sqrt{z-1}
&=\sqrt a\sqrt{\frac{x-1}{a}}+\sqrt b\sqrt{\frac{y-1}{b}}+\sqrt c\sqrt{\frac{z-1}{c}}\\
&\leq\sqrt{(a+b+c)\left(\frac{x-1}{a}+\frac{y-1}{b}+\frac{z-1}{c}\right)}.
\end{aligned}
\]
Now comes the crucial move: take $a=x$, $b=y$, $c=z$. Q.E.D.!
% Source page 36 - right
\paragraph{Kvant: Vasile C\^irtoaje.}
Prove that for any real numbers $a,b,c,x,y,z$,
\[ax+by+cz+\sqrt{(a^2+b^2+c^2)(x^2+y^2+z^2)}
\geq\frac23(a+b+c)(x+y+z).\]
The usual temptation is to apply Cauchy to the big square-root term, but that is just a wild-goose chase.
Rather, as usual, let us do something unusual.
\[
\begin{aligned}
\frac23(a+b+c)(x+y+z)-(ax+by+cz)
&=\sum_{\mathrm{cyc}}a\left(\frac{2y+2z-x}{3}\right)\\
&\leq\sqrt{a^2+b^2+c^2}\,
\sqrt{\sum_{\mathrm{cyc}}\left(\frac{2y+2z-x}{3}\right)^2}.
\end{aligned}
\]
You know what? We have thus reached an unusual fact. Look below!
\[\sum_{\mathrm{cyc}}\left(\frac{2x+2y-z}{3}\right)^2=\sum x^2.\]
Wow! A.T.: always true. Q.E.D.!

\paragraph{A problem that took Titu Andreescu several weeks to resolve.}
For positive reals $x,y,z,a,b,c$ such that $xy+yz+zx=3$, prove that
\[\frac a{b+c}(y+z)+\frac b{c+a}(z+x)+\frac c{a+b}(x+y)\geq3.\]
It is sufficient to establish
\[\frac a{b+c}(y+z)+\frac b{c+a}(z+x)+\frac c{a+b}(x+y)
\geq\sqrt{3(xy+yz+zx)}.\]
We may assume that $x+y+z=1$. How?
Our inequality takes the form
\[\frac a{b+c}+\frac b{c+a}+\frac c{a+b}
\geq\sqrt{3(xy+yz+zx)}+\frac a{b+c}x+\frac b{c+a}y+\frac c{a+b}z.\]
By Cauchy,
\[\frac a{b+c}x+\frac b{c+a}y+\frac c{a+b}z
\leq\sqrt{\sum_{\mathrm{cyc}}\left(\frac a{b+c}\right)^2
\sum_{\mathrm{cyc}}x^2}.\]
We must now prove
\[\sqrt{\sum\left(\frac a{b+c}\right)^2\sum x^2}
+\sqrt{3(xy+yz+zx)}
\mathrel{\stackrel{?}{\leq}}\sum\frac a{b+c}.\]
Since $\sqrt{12ABC}\leq(3B+4AC)/2$, we get
\[
\begin{aligned}
\sqrt{12ABC}+AB+3C&\leq\frac{3B}{2}+2AC+AB+3C,\\
(\sqrt{AB}+\sqrt{3C})^2&\leq\frac32B+2AC+AB+3C
=\left(\frac32+A\right)(B+2C).
\end{aligned}
\]
Put $A=\sum(a/(b+c))^2$, $B=\sum x^2$, and $C=\sum xy$. Then
\[
\sqrt{\sum\left(\frac a{b+c}\right)^2\sum x^2}+\sqrt{3(xy+yz+zx)}
\leq\sqrt{\left(\frac32+\sum\left(\frac a{b+c}\right)^2\right)
\left(\sum x^2+2\sum xy\right)}.
\]
Thus it remains to prove
\[
\frac32+\sum\left(\frac a{b+c}\right)^2
\mathrel{\stackrel{?}{\leq}}\left(\sum\frac a{b+c}\right)^2.
\]
Expanding the right-hand side reduces this to
\[\sum_{\mathrm{cyc}}\frac{ab}{(b+c)(c+a)}\mathrel{\stackrel{?}{\geq}}\frac34.\]
In full,
\[
\frac{ab}{(b+c)(c+a)}+\frac{bc}{(c+a)(a+b)}+\frac{ca}{(a+b)(b+c)}
\mathrel{\stackrel{?}{\geq}}\frac34
\]
is equivalent to
\[
ab(a+b)+bc(b+c)+ca(c+a)\geq6abc,
\]
which follows from AM--GM applied to the six terms. True! W$^5$; Q.E.D.

A close inspection justifies the normalization above.
We have $xy+yz+zx=3$. Let $x+y+z=k$, and put
$X=x/k$, $Y=y/k$, and $Z=z/k$. Then $X+Y+Z=1$.
Divide the homogeneous inequality by $k$ to obtain
\[
\frac a{b+c}(Y+Z)+\frac b{c+a}(Z+X)+\frac c{a+b}(X+Y)
\geq\sqrt{3(XY+YZ+ZX)}.
\]

\paragraph{A square-root heuristic: China (1982).}
Let $n\geq2$ and let $a_1,a_2,\ldots,a_n$ be positive numbers such that $a_1+\cdots+a_n=1$.
Prove that
\[\frac{a_1}{\sqrt{1-a_1}}+\cdots+\frac{a_n}{\sqrt{1-a_n}}
\geq\frac1{\sqrt{n-1}}(\sqrt{a_1}+\cdots+\sqrt{a_n}).\]
By square-root heuristics,
\[\sum_{\mathrm{cyc}}\sqrt{a_i}\leq\sqrt{n\sum a_i}=\sqrt n,
\qquad\frac1{\sqrt{n-1}}\sum_{\mathrm{cyc}}\sqrt{a_i}\leq\sqrt{\frac n{n-1}}.\]
We seek $\sum a_i/\sqrt{1-a_i}\geq\sqrt{n/(n-1)}$.
Chebyshev's inequality gives
\[\sum_{\mathrm{cyc}}\frac{a_i}{\sqrt{1-a_i}}
\geq\frac1n\sum a_i\sum\frac1{\sqrt{1-a_i}}
=\frac1n\sum\frac1{\sqrt{1-a_i}}.\]
Therefore it is sufficient to prove
$\frac1n\sum1/\sqrt{1-a_i}\geq\sqrt{n/(n-1)}$.
By Cauchy--Schwarz,
\[\sum\frac1{\sqrt{1-a_i}}\geq\frac{n^2}{\sum\sqrt{1-a_i}}
\mathrel{\stackrel{?}{\geq}}\frac{n\sqrt n}{\sqrt{n-1}}.\]
The latter reduces to $\sqrt{n(n-1)}\geq\sum\sqrt{1-a_i}$.
By square-root heuristics,
\[\sum_{\mathrm{cyc}}\sqrt{1-a_i}\leq\sqrt{n\sum_{\mathrm{cyc}}(1-a_i)}=\sqrt{n(n-1)}.\]
Q.E.D.
\paragraph{A two-case AM--GM argument.}
For $x,y,z\in\mathbb R$,
\[x^2+y^2+z^2\geq x\sqrt{y^2+z^2}+y\sqrt{x^2+z^2}.\]
Indeed, AM $\geq$ GM gives
\[
\begin{aligned}
x^2+y^2+z^2&=x^2+\left(\sqrt{y^2+z^2}\right)^2
\geq2x\sqrt{y^2+z^2}, &&\text{I},\\
x^2+y^2+z^2&=y^2+\left(\sqrt{x^2+z^2}\right)^2
\geq2y\sqrt{x^2+z^2}, &&\text{II}.
\end{aligned}
\]
If $x\sqrt{y^2+z^2}\geq y\sqrt{x^2+z^2}$, substitute in I and the result follows.
But that inequality need not hold: if $y\sqrt{x^2+z^2}\geq x\sqrt{y^2+z^2}$, substitute in II.
The conclusion follows.

\paragraph{Clever AM--GM: AMTI inter-final.}
Let $a,b,c,d>0$ satisfy
$\dfrac1{1+a^4}+\dfrac1{1+b^4}+\dfrac1{1+c^4}+\dfrac1{1+d^4}=1$.
Prove $abcd\geq3$.
\paragraph{Solution 1, due to Pranjal Warade.}
\[
\frac1{1+a^4}+\frac1{1+b^4}+\frac1{1+c^4}
=1-\frac1{1+d^4}=\frac{d^4}{1+d^4}
\geq3\sqrt[3]{\frac1{(1+a^4)(1+b^4)(1+c^4)}}.
\]
Applying the analogous inequalities cyclically and multiplying them gives $(abcd)^4\geq3^4$, hence $abcd\geq3$.
% Source page 14 - right
\paragraph{Solution 2, due to the author.}
Put $A=1/(1+a^4)$, $B=1/(1+b^4)$, $C=1/(1+c^4)$, $D=1/(1+d^4)$.
Then $a^4=(1-A)/A$, and similarly for the others.
Using $A+B+C+D=1$,
\begin{align*}
\frac{1-A}{A}\frac{1-B}{B}\frac{1-C}{C}\frac{1-D}{D}
&=\frac{B+C+D}{A}\frac{A+C+D}{B}\frac{A+B+D}{C}\frac{A+B+C}{D}\\
&\geq\frac{3(BCD)^{1/3}\,3(ACD)^{1/3}\,3(ABD)^{1/3}\,3(ABC)^{1/3}}{ABCD}
=3^4.
\end{align*}
Thus $abcd\geq3$.

\subsection{Right or wrong: who cares!}
For positive $a,b,c$, prove
\[
\sum_{\mathrm{cyc}}\frac{a^4+b^2}{c^3+a^2b}\geq3.
\]
First claim that the minimum of the left-hand side is $k$.
Now $\max(k)$ is the minimum of the left-hand side, so let us compute $\max(k)$:
\begin{align*}
\sum_{\mathrm{cyc}}\frac{a^4+b^2}{c^3+a^2b}
&\geq\sum_{\mathrm{cyc}}\frac{2a^2b}{c^3+a^2b}
=\sum_{\mathrm{cyc}}\frac1{c^3/(2a^2b)+1/2}\\
&\geq\frac{18/2}{\sum_{\mathrm{cyc}}c^3/(2a^2b)+3/2}\geq k.
\end{align*}
The final comparison with $k$ is not justified: the product of the three
quantities $c^3/(a^2b)$ is $1$, so AM--GM gives

\paragraph{An inequality from the constraint.}
For real nonzero $a,b,c$ satisfying
\[
\sum_{\mathrm{cyc}}\frac{ab}{c^2}
=3\sum_{\mathrm{cyc}}\frac1{a^2}=3,
\]
prove
\[
\sum_{\mathrm{cyc}}\frac{a^2}{b^2c^2}\geq\frac{11}{8}.
\]
As written, this is false: $a=b=c=\sqrt3$ satisfies the constraint, while
the left-hand side is $1$. The calculation does prove the result on the
non-equal branch of the constraint. First focus on the cleavage of the constraint!
\begin{align*}
\sum_{\mathrm{cyc}}\frac{ab}{c^2}=3
&\Longrightarrow\sum_{\mathrm{cyc}}(ab)^3-3(abc)^2=0\\
&\Longrightarrow\frac12\left(\sum_{\mathrm{cyc}}ab\right)
\left(\sum_{\mathrm{cyc}}(ab-bc)^2\right)=0\\
&\Longrightarrow\frac{a^3b^3c^3}2
\left(\sum_{\mathrm{cyc}}\frac1a\right)
\left(\sum_{\mathrm{cyc}}\left(\frac1a-\frac1b\right)^2\right)=0.
\end{align*}
If $a,b,c$ are not all equal, we therefore have $\sum1/a=0$. This also follows from
\[
\det\begin{pmatrix}
bc&ca&ab\\
ca&ab&bc\\
ab&bc&ca
\end{pmatrix}=0.
\]
We also have $\sum1/a^2=1$. Put $x=1/a$, $y=1/b$, $z=1/c$.
Then
\[
x+y+z=0,\qquad x^2+y^2+z^2=1,qquad xy+yz+zx=-\frac12.
\]
Let us do some childish computations. We have
\[
\sum x^2y^2=(\sum xy)^2-2xyz\sum x=\frac14.
\]
Moreover,
\[
(\sum x^2y^2)^2
=\sum x^4y^4+2x^2y^2z^2\sum x^2
=\sum x^4y^4+2x^2y^2z^2.
\]
We further need $x^2y^2z^2\leq1/54$. To see this, two of $x,y,z$
have the same sign; write their absolute values as $u,v$, so the absolute
value of the third is $u+v$. If $s=u+v$ and $t=uv$, then
\[
2(s^2-t)=1,qquad t\leq\frac{s^2}{4},qquad
\frac12\leq s^2\leq\frac23.
\]
Hence
\[
x^2y^2z^2=t^2s^2=s^2\left(s^2-\frac12\right)^2\leq\frac1{54}.
\]
It follows that
\[
\sum x^4y^4=\frac1{16}-2x^2y^2z^2
\geq\frac{11}{16\cdot27}.
\]
Now $a^2b^2c^2\geq54$ and
$\sum1/(a^4b^4)\geq11/(16\cdot27)$, so
\[
\sum_{\mathrm{cyc}}\frac{a^2b^2c^2}{a^4b^4}
\geq\frac{11\cdot54}{16\cdot27}
\Longrightarrow
\sum_{\mathrm{cyc}}\frac{a^2}{b^2c^2}\geq\frac{11}8.
\]
That is what I am talking about! Q.E.D.!
