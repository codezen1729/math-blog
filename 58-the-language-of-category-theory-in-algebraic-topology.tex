\setcounter{definition}{0}\begin{definition}[Category]
A category $\mathcal C$ consists of a class of objects, denoted by
$\Ob(\mathcal C)$, and a class of composable morphisms, denoted by
$\Mor(\mathcal C)$.
\end{definition}
\begin{figure}[H]
 \centering\includegraphics[width=.55\textwidth]{m1-fig-02.pdf}
 \FigureTag{M1.2}{fig:m1-fig-02}
\end{figure}

% Source page 2
\setcounter{example}{0}\begin{example}
$\cat{Sets}$, $\cat{Grps}$, $\cat{Rings}$, $\cat{Top}$,
$k\text{-}\cat{Top}$, $\cat{Top}_*$, $k\text{-}\cat{Top}_*$,
$k\text{-}\cat{Top}_*/{\sim}$, $\cat{CRings}$, $\cat{Man}$, $\ldots$.
\end{example}
\setcounter{definition}{1}\begin{definition}[Functor]
For categories $\mathcal C$ and $\mathcal D$, a functor
$F\colon\mathcal C\to\mathcal D$ gives maps
$F\colon\Ob(\mathcal C)\to\Ob(\mathcal D)$ and
$F\colon\Mor(\mathcal C)\to\Mor(\mathcal D)$ such that
\[
 F(*\xrightarrow{f}*)=\bigl(F(*)\xrightarrow{F(f)}F(*)\bigr),\qquad
 F(\id_*)=\id_{F(*)},\qquad F(f\circ g)=F(f)\circ F(g).
\]
\end{definition}
\setcounter{example}{1}\begin{example}
Let $\cat{PCTop}_*$ be the category of pointed path-connected topological spaces and base-point-preserving maps.
The assignments $X\mapsto\pi_1(X)$ (the fundamental group of $X$) and
\[
 (h\colon X\to Y)\longmapsto
 \bigl([f]\xmapsto{h_*}[h\circ f]\colon\pi_1(X)\to\pi_1(Y)\bigr)
\]
give a functor from $\cat{PCTop}_*$ to $\cat{Grps}$. Moreover,
$\mathbb C\setminus\{0\}\mapsto\mathbb Z$, $\partial\mathbb D\mapsto\mathbb Z$,
and $z\mapsto z^2$ maps to $n\mapsto2n$.
\end{example}
Functors help us better understand relations in a category
by translating them to a different category. Given two such translations,
is there a way not to get lost in translation?
\setcounter{definition}{2}\begin{definition}[Natural transformation]
Let $F,G\colon\mathcal C\to\mathcal D$ be two functors. A natural
transformation is a collection of morphisms
$\{\nu_A\}_{A\in\Ob(\mathcal C)}$ such that the following square commutes.
\begin{figure}[H]
 \centering\includegraphics[width=.78\textwidth]{m1-fig-03.pdf}
 \FigureTag{M1.3}{fig:m1-fig-03}
\end{figure}
We say that $\{\nu_A\}_{A\in\Ob(\mathcal C)}$ is a \emph{natural isomorphism}
if $\nu_A$ is an isomorphism for every $A\in\Ob(\mathcal C)$.
\end{definition}

% Source page 3
\paragraph{Principle.}
Every statement in category theory has a dual statement obtained by replacing
$f\circ g$ with $g\circ f$ and domain with codomain, essentially flipping the
arrows of the morphisms.

Before turning to examples, there is a small set-theoretic point. Certain collections, such as the collection of all
groups or of all singletons, cannot be sets. This motivates the language of classes,
whose elements are sets satisfying specified properties. For our purposes, we restrict attention to
small or locally small categories, thereby avoiding the set-versus-class
conundrum.
\subsection{Some Pillars of Category Theory}
\subsubsection{Functor Categories}
Let $\mathcal C=\Fun(\mathcal A,\mathcal B)$, where $\mathcal A$ and
$\mathcal B$ are categories. Its objects are functors
$F\colon\mathcal A\to\mathcal B$, and its morphisms are natural transformations.
We say $\mathcal C\cong\mathcal D$ if there exist
$F\colon\mathcal C\to\mathcal D$ and $G\colon\mathcal D\to\mathcal C$ such that
$F\circ G=\id_{\mathcal D}$ and $G\circ F=\id_{\mathcal C}$.
We say $\mathcal C\sim\mathcal D$ if
$F\circ G\sim\id_{\mathcal D}$ and $G\circ F\sim\id_{\mathcal C}$,
via natural isomorphisms.
\subsubsection{Representable Functors}
For $X,C\in\Ob(\mathcal C)$, define
$h_X\colon\mathcal C^{\mathrm{op}}\to\cat{Sets}$ by
% Source page 4 (the defining formula overlaps the previous page boundary).
\[
 h_X(C)=\Hom(C,X),\qquad
 h_X(f\colon C\to C')\colon\Hom(C',X)\to\Hom(C,X),\quad g\mapsto g\circ f.
\]
We say that a functor $F\colon\mathcal C^{\mathrm{op}}\to\cat{Sets}$ is
\emph{representable} if there exists $C\in\Ob(\mathcal C)$ such that $F\cong h_C$.

Many features of a category can be studied through representable functors.
This is akin to studying $\mathcal C$ through its action on itself via
$\Hom_{\mathcal C}(-,-)$.
\subsubsection{Final and Initial Objects}
Let $F\colon\mathcal C^{\mathrm{op}}\to\cat{Sets}$ be given by
$F(C)=*$ (a singleton) and $F(f)=\id_*$. We say $I$ is a \emph{final object}
of $\mathcal C$ if $h_I\cong F$, where $\cong$ denotes natural isomorphism.
\paragraph{Construction of $I$.}
Let $I\in\Ob(\mathcal C)$ be such that, for every $C\in\Ob(\mathcal C)$,
there is a unique $f\in\Mor(\mathcal C)$ with $C\xrightarrow{f}I$.
\begin{claim}
$h_I\cong F$.
\end{claim}
For every $C\in\mathcal C$, the sets $h_I(C)$ and $F(C)$ are isomorphic,
as they both have a single element. An \emph{initial object} is the dual notion
of a final object. Therefore final objects represent the constant singleton functor;
categories with no final object yield standard examples in which this functor is non-representable.
% Source page 5 (opening line continues the preceding page).
\subsubsection{Limits and Colimits}
Let $\chi\colon J\to\mathcal C$ be a functor, where $J$ is often called the
\emph{index category}. Let $C_A\colon J\to\mathcal C$ be given by
$C_A(j)=A$ for every $j\in\Ob(J)$ and $C_A(\theta)=\id_A$ for every
$\theta\in\Mor(J)$. Define
\[
 F_\chi\colon\mathcal C^{\mathrm{op}}\to\cat{Sets},\qquad
 F_\chi(A):=\Nat(C_A,\chi),
\]
where $\Nat$ stands for natural transformations, and
\[
 \begin{aligned}
 F_\chi(f\colon A\to B)&\colon\Nat(C_B,\chi)\to\Nat(C_A,\chi),\\
 \{\nu_i\}_{i\in\Ob(J)}&\longmapsto
 \{\nu_i\circ f\}_{i\in\Ob(J)}.
 \end{aligned}
\]
If $F_\chi\cong h_C$, then $C=\lim(\chi)$ is called the \emph{limit} of $\chi$.
\paragraph{Construction of $C$.}
Define the category $\Cone(\chi)$, with
\[
 \Ob(\Cone(\chi))=
 \left(C,\{f_i\colon C\to\chi(i)\}_{i\in J}\right),\qquad C\in\Ob(\mathcal C),
\]
where the $f_i$ are such that, if $\alpha\colon i\to j$, then the following
diagram commutes.

\begin{figure}[H]
 \centering\includegraphics[width=.55\textwidth]{m1-fig-04.pdf}
 \FigureTag{M1.4}{fig:m1-fig-04}
\end{figure}
The morphisms are
\[
 \Mor(\Cone(\chi))=
 \left(f\colon(C,\{f_i\}_i)\longrightarrow(C',\{f_i'\}_i)\right),
\]
where $f$ is such that the following diagram commutes for every $i\in\Ob(J)$.
\begin{figure}[H]
 \centering\includegraphics[width=.55\textwidth]{m1-fig-05.pdf}
 \FigureTag{M1.5}{fig:m1-fig-05}
\end{figure}
$\Cone(\chi)$ is called the \emph{cone category} of $\chi$.
A final object of $\Cone(\chi)$ is called the limit of $\chi$.
Let $\lim(\chi)$ be the final object. Then it is easy to see that
$h_{\lim(\chi)}\cong F_\chi$, namely,
\[
 \Hom(D,\lim(\chi))\cong\Nat(C_D,\chi)\qquad\text{(as sets)}.
\]
Given $f\colon D\to\lim(\chi)$, we obtain
$\bigl(D,\{\chi_i\circ f\}_{i\in\Ob(J)}\bigr)$
% Source page 7
in $\Ob(\Cone(\chi))$, where $\chi_i\colon\lim(\chi)\to\chi(i)$.
By definition, $f$ must be unique, so we have the natural map
$f\mapsto\{\chi_i\circ f\}_{i\in\Ob(J)}$.
A \emph{colimit} is the dual notion of a limit in $\mathcal C$.
\subsubsection{Adjunctions (or Adjoints)}
Let $\mathcal C$ and $\mathcal D$ be two categories, and let
$F\colon\mathcal C\to\mathcal D$ be a functor. An adjoint to $F$ is a
functor $G\colon\mathcal D\to\mathcal C$ such that
\[
 \Hom_{\mathcal C}(C,G(D))\cong\Hom_{\mathcal D}(F(C),D).
\]
This can be rephrased in terms of a representable functor. Define
\[
 \overline F\colon\mathcal C^{\mathrm{op}}\to\cat{Sets},\qquad
 C\longmapsto\Hom_{\mathcal D}(F(C),D),
\]
% Source page 8
and send $f\colon C\to C'$ to the map
\[
 \bigl(g\colon F(C')\to D\bigr)\longmapsto
 \bigl(g\circ F(f)\colon F(C)\to D\bigr).
\]
Suppose $\overline F$ is representable. Then there exists $\overline D$ such that
\[
 \Hom_{\mathcal C}(C,\overline D)\cong\Hom_{\mathcal D}(F(C),D).
\]
Set $G(D):=\overline D$. The action of $G$ on morphisms is induced by naturality of the representing isomorphism.
\subsection{Constructing New Categories from Old}
\subsubsection{Cone Category}
The cone construction above is one example. Objects in a cone
category generalize the notion of a morphism. A morphism relates a particular
object to another, but a cone object relates to a collection of objects that are already related to one another.

\subsubsection{Quotient Category}
Let $\mathcal C$ be a category. For every $X,Y\in\Ob(\mathcal C)$, an
equivalence relation $\sim$ is set up on $\Hom_{\mathcal C}(X,Y)$ such that
\[
 \left.\begin{gathered}
 f_1\sim f_2\quad\text{in }\Hom_{\mathcal C}(X,Y),\\
 g_1\sim g_2\quad\text{in }\Hom_{\mathcal C}(Z,X)
 \end{gathered}\right\}
 \quad\Longrightarrow\quad f_1\circ g_1\sim f_2\circ g_2.
\]
Also, $X\sim Y$ if there exist $f\colon X\to Y$ and $g\colon Y\to X$ such that
$f\circ g\sim\id_Y$ and $g\circ f\sim\id_X$.
% Source page 10
Then $\mathcal C/{\sim}$ is the \emph{quotient category}.
\setcounter{example}{2}\begin{example}
$\Fun(\mathcal C,\mathcal D)/{\sim}$, for natural isomorphism;
$\cat{Top}/{\sim}$, for homotopy equivalence.
\end{example}
\subsubsection{Comma Category}
Let $\mathcal C$ be a given category and $B\in\Ob(\mathcal C)$.
Write $\mathcal C\downarrow B$ for the \emph{comma category over $B$}.
Its objects are maps in $\mathcal C$, of the form $X\xrightarrow{\pi}B$
(a ``$B$-parametrized family of objects in $\mathcal C$'').
The set
\[
 \Hom_{\mathcal C\downarrow B}
 \bigl((X\xrightarrow{\pi}B),(X'\xrightarrow{\pi'}B)\bigr)
\]
consists of the maps $f\colon X\to X'$ for which the following diagram commutes.

 \FigureTag{M1.7}{fig:m1-fig-07}
\end{figure}

\setcounter{example}{3}\begin{example}
If $\mathcal C=\cat{Top}$ and $B=\{*\}$, then
$B\downarrow\mathcal C=\cat{Top}_*$ (based spaces).
\end{example}

\subsubsection{Category of Presheaves}
Given a category $\mathcal C$, define $\PSh(\mathcal C)$ by taking functors
$F\colon\mathcal C^{\mathrm{op}}\to\cat{Sets}$ as objects and natural
transformations as morphisms. Thus
$\PSh(\mathcal C)=\Fun(\mathcal C^{\mathrm{op}},\cat{Sets})$.
\subsection{Standard Instances}
\subsubsection{Morphisms as Functors}
Let $f\colon A\to B$ be a morphism, with $A,B\in\Ob(\mathcal C)$.
Let $J$ be some index category and let $\chi_A\colon J\to\mathcal C$ be the
functor given by
\[
 \chi_A(i)=A\quad(i\in\Ob(J)),\qquad
 \chi_A(f\colon i\to j)=\id_A.
\]
Then $\Nat(\chi_A,\chi_B)$ captures the morphisms from $A$ to $B$, and
each element of $\Nat(\chi_A,\chi_B)$ is a morphism in $\Fun(J,\mathcal C)$.
\subsubsection{Objects as Morphisms}
Let $X\in\Ob(\mathcal C)$. The points of $X$ can be thought
% Source page 12
of as $\Hom_{\mathcal C}(*,X)$, where $*$ is a final object in $\mathcal C$.
``Generalized points'' can be thought of as $\Hom_{\mathcal C}(A,X)$.
We are simply generalizing these notions from $\mathcal C=\cat{Sets}$,
where $*$ is a singleton set, and morphisms from $*$ to $X$ in $\cat{Sets}$
give a correspondence with points of $X$.
Similarly, morphisms can be treated as objects once realized as functors in
the functor category.
\subsubsection{Groups as Categories}
Given a group $G$, define a category $\mathcal BG$ as follows:
\[
 \Ob(\mathcal BG):=\{*\}\quad\text{(a singleton)},\qquad
 \Mor(\mathcal BG):=G,
\]
where $g\in G$ is a morphism $*\to*$ and multiplication is composition.
\begin{remark}
If $G$ is nontrivial, $\mathcal BG$ has no final object.
\end{remark}
\subsubsection{Forgetful Functors}
Let $\mathcal C$ be a category of certain sets with a particular
% Source page 13
structure imposed. The forgetful functor is
\[
 U\colon\mathcal C\to\cat{Sets},\qquad
 U(C)=C\quad\text{(the underlying set)},\qquad
 U(f\colon C\to D)=f\colon C\to D\quad\text{(the set map)}.
\]
\subsubsection{Exponentials}
Let $C,B\in\Ob(\mathcal C)$. The \emph{exponential} is an object $C^B$
such that, for any $f\colon A\times B\to C$, there exists a unique
$\overline f\colon A\to C^B$ for which the following diagram commutes.

\subsubsection{Group Objects in Categories}
% Source page 14
Let $\mathcal C$ be a category closed under finite products, and let
$G\in\mathcal C$ be an object. The structure of a group on $G$ is the data
of morphisms
\[
 \mu\colon G\times G\to G,\qquad \nu\colon G\to G,\qquad e\colon *\to G,
\]
where $*$ is a terminal object, such that the following diagrams commute.
\[
\begin{tikzcd}[row sep=large,column sep=large,cells={nodes={font=\normalsize}}]
G\times G\times G \arrow[r,"\mu\times\id_G"] \arrow[d,"\id_G\times\mu"']
  &G\times G\arrow[d,"\mu"]\\
G\times G\arrow[r,"\mu"']&G
\end{tikzcd}
\]
\[
\begin{tikzcd}[row sep=large,column sep=large,cells={nodes={font=\normalsize}}]
G\cong *\times G \arrow[r,"e\times\id_G"] \arrow[dr,"\id_G"']
  &G\times G\arrow[d,"\mu"]
  &&G\cong G\times * \arrow[r,"\id_G\times e"] \arrow[dr,"\id_G"']
  &G\times G\arrow[d,"\mu"]\\
  &G&&&G
\end{tikzcd}
\]
\[
\begin{tikzcd}[row sep=large,column sep=large,cells={nodes={font=\normalsize}}]
G \arrow[r,"{(\id_G,\nu)}"] \arrow[d,"!"']
  &G\times G\arrow[d,"\mu"]
  &&G \arrow[r,"{(\nu,\id_G)}"] \arrow[d,"!"']
  &G\times G\arrow[d,"\mu"]\\
*\arrow[r,"e"']&G&&*\arrow[r,"e"']&G
\end{tikzcd}
\]
\setcounter{example}{4}\begin{example}
In $\cat{Sets}$, a group object is a group; in $\cat{Top}$, it is a topological
group; in $\cat{Man}$, it is a Lie group. In $k\text{-}\cat{Top}_*/{\sim}$,
it is an $H$-space; the ambient category is described below.
\end{example}
\subsection{Certain Limit Constructions in Algebraic Topology}
\subsubsection{Products and Coproducts}
Let $\{X_i\}_{i\in I}$ be objects in $\mathcal C$. Consider the discrete
category $X$, with $\Ob(X)=X_i$ and $\Mor(X)=\id_{X_i}$, and let
$\chi\colon X\hookrightarrow\mathcal C$ be the inclusion.
The limit and colimit of $\chi$, if they exist, are products and coproducts.
In $\cat{Top}$, the product is $\prod_{i\in I}X_i$, with the product topology,
and the coproduct is $\coprod_{i\in I}X_i$.
In $*\downarrow\cat{Top}\sim\cat{Top}_*$, the product of the $(X_i,x_i)$ is
$\bigl(\prod_{i\in I}X_i,\{x_i\}_{i\in I}\bigr)$ and the coproduct is
$\bigvee_{i\in I}(X_i,x_i)$.
\subsubsection{Joins ($\mathcal C=\cat{Top}$)}
% Source page 16
Given $X,Y\in\mathcal C$, the join of $X$ and $Y$, denoted $X*Y$, is the
colimit of the following diagram.
\begin{figure}[H]
 \centering\includegraphics[width=.94\textwidth]{m1-fig-13.pdf}
 \FigureTag{M1.13}{fig:m1-fig-13}

Here $\pi_1(x,y)=x$, $i_1(x,y)=(x,y,0)$, $i_2(x,y)=(x,y,1)$, and
$\pi_2(x,y)=y$. Explicitly,
\[
 X*Y=(X\times Y\times[0,1])/{\sim},\qquad
 \begin{cases}
 (x,y,0)\sim(x,y',0),\\
 (x,y,1)\sim(x',y,1),
 \end{cases}
\]
which is essentially all the line segments joining points in $X$ and $Y$.
The special cases are
\[
 \begin{aligned}
 X*\{\mathrm{pt}\}&=: \Cone(X)\cong(X\times I)/(X\times\{0\})
 &&\text{(denoted $CX$)},\\
 X*S^0&=: \operatorname{Suspension}(X)
 \cong(X\times I)/(X\times\{0\},\,X\times\{1\})
 &&\text{(denoted $\Sigma X$)}.
 \end{aligned}
\]
Also $\Sigma S^n\cong S^{n+1}$. Therefore a purely category-theoretic way
of constructing an $n$-sphere would be to take
$S^n=\underbrace{S^0*\cdots*S^0}_{n+1\text{ times}}$.
\subsubsection{Pushouts and Pullbacks}
A pushout is a colimit of the first diagram below. The second gives the example
$\Cone(f):=(X\amalg CA)/{\sim}$, where $(a,1)\sim f(a)$.
\begin{figure}[H]

The dual notion is a pullback.
\begin{remark}
The limit and colimit of a diagram are essentially two ways of putting
objects together such that the relations between them are preserved.
\end{remark}
In abelian groups, putting together
$G_0\xrightarrow{f_0}G_1\xrightarrow{f_1}G_2\xrightarrow{f_2}\cdots$
would mean looking at $(x_0,x_1,\ldots)\in\prod_{i=0}^{\infty}G_i$ such
that $f_i(x_i)=x_{i+1}$ (the direct product).
In $\cat{Top}$, putting together $A\xrightarrow{f}X$ and $A\xrightarrow{g}Y$
would give $(X\amalg Y)/\{f(x)\sim g(x):x\in A\}$.
In $\cat{Grps}$, putting together the same diagram would give
\[
 (X*Y)/N\bigl(f(a)g(a)^{-1}:a\in A\bigr),
\]
the amalgamated product.
In a category $\mathcal C$, limits and colimits play the roles of $\prod$ and $\coprod$ while preserving the relations in the diagram.

We now arrive at the central lemma of category theory.
\setcounter{theorem}{0}\begin{lemma}[Yoneda's lemma]\label{lem:m1-yoneda}
Let $F\in\PSh(\mathcal C)$. For any $X\in\Ob(\mathcal C)$, there is an
isomorphism of sets
\[
 \Hom_{\PSh(\mathcal C)}(h_X,F)\cong F(X)
\]
that is natural in both $X$ and $F$.
\end{lemma}
% Source page 18: blank.
% Source page 19
\begin{proof}[Sketch of proof]
Suppose $\theta\in\Hom_{\PSh(\mathcal C)}(h_X,F)$, i.e.
$\{\theta_A\colon h_X(A)\to F(A)\}_{A\in\Ob(\mathcal C)}$.
Define
\[
 \Phi\colon\Hom_{\PSh(\mathcal C)}(h_X,F)\to F(X),\qquad
 \Phi\bigl(\{\theta_A\}_{A\in\Ob(\mathcal C)}\bigr)=\theta_X(\id_X).
\]
Let $a\in F(X)$. Then $\Psi(a)\in\Hom_{\PSh(\mathcal C)}(h_X,F)$ is given by
\[
 \Psi(a)_A(f):=F(f)(a)\qquad(f\in h_X(A)),
\]
the natural evaluation. Naturality and the two inverse identities follow directly from the definitions.
\end{proof}
We conclude with two applications of Yoneda's lemma.
\subsubsection{Application 1}
If $\mathcal C$ is Cartesian closed, then for any three objects $X,Y,Z$,
\[
\Hom_{\mathcal C}(X,Z^Y)
 \cong\Hom_{\mathcal C}(X\times Y,Z),
\]
often written as $Z^{X\times Y}\cong(Z^Y)^X$.
$\cat{Top}$ is not Cartesian closed. Hence one considers
$k\text{-}\cat{Top}$, a full subcategory of $\cat{Top}$ consisting of
compactly generated spaces. Using this, in
$k\text{-}\cat{Top}_*/{\sim}\cong(*\downarrow k\text{-}\cat{Top})/{\sim}$,
one can prove that
% Source page 20
\[
 \begin{aligned}
 \Hom_{(k\text{-}\cat{Top}_*)/{\sim}}(X,\Omega_{y_0}Y)
 &:=\pi_b(X,\Omega_{y_0}Y)\\
 &\cong\pi_b(\Sigma X,Y)
 =\Hom_{(k\text{-}\cat{Top}_*)/{\sim}}(\Sigma X,Y).
 \end{aligned}
\]
Here $\pi_b$ denotes the collection of based continuous maps, up to based homotopy.
The homotopy sets are
\[
 \begin{aligned}
 \text{zeroth homotopy set}&:=\pi_b(S^0,Y),\\
 \text{first homotopy set}&:=\pi_b(S^0,\Omega_{y_0}Y),\\
 &\ \vdots\\
 \text{$n$th homotopy set}&:=\pi_b(S^0,\Omega^n(Y))
 \cong\pi_b(\Sigma^nS^0,Y)\cong\pi_b(S^n,Y).
 \end{aligned}
\]
\subsubsection{Application 2}
Let $\mathcal C$ be a category and $X\in\Ob(\mathcal C)$. Then
$X$ is a group object if and only if $h_X\colon\mathcal C^{\mathrm{op}}\to\cat{Sets}$
lifts to groups, i.e. there exists $\widetilde h_X$ making the following
diagram commute; $U$ is the forgetful functor.
\begin{figure}[H]
 \centering\includegraphics[width=.62\textwidth]{m1-fig-15.pdf}
 \FigureTag{M1.15}{fig:m1-fig-15}
\end{figure}
Now $\Omega_{y_0}Y$ can be made into a group object in
$(k\text{-}\cat{Top}_*)/{\sim}$, i.e. an $H$-space.
By Yoneda's lemma, one can give a group structure to
\[
 \Hom_{(k\text{-}\cat{Top}_*)/{\sim}}(A,\Omega_{y_0}Y)
 =\pi_b(A,\Omega_{y_0}Y)
\]
for every $A\in\Ob((k\text{-}\cat{Top}_*)/{\sim})$, since the functor
$\pi_b(-,\Omega_{y_0}Y)$ lifts to $\cat{Grps}$.
This way one can give higher homotopy sets a group structure. Homology and
cohomology are likewise group-valued functors, but their group structures come
from addition of chains and cochains rather than from this loop-space argument.
